### 📝 Commandes Unix Implémentées (14)


| Commande | Description                               | Exemple                     |
| -------- | ----------------------------------------- | --------------------------- |
| `cat`    | Affiche le contenu d'un fichier           | `cat fichier.txt`           |
| `cd`     | Change de répertoire                      | `cd docs/../config`         |
| `clear`  | Efface l'écran du terminal                | `clear`                     |
| `cp`     | Copie un fichier                          | `cp source.txt dest.txt`    |
| `echo`   | Affiche du texte ou écrit dans un fichier | `echo "Hello" > test.txt`   |
| `exit`   | Quitte l'application                      | `exit`                      |
| `help`   | Affiche l'aide de toutes les commandes    | `help` ou `help ls`         |
| `ls`     | Liste les fichiers/dossiers               | `ls -l`                     |
| `mkdir`  | Crée un nouveau dossier                   | `mkdir nouveau_dossier`     |
| `mv`     | Déplace ou renomme                        | `mv ancien.txt nouveau.txt` |
| `pwd`    | Affiche le chemin du répertoire courant   | `pwd`                       |
| `rm`     | Supprime un fichier                       | `rm fichier.txt`            |
| `rmdir`  | Supprime un dossier **vide**              | `rmdir dossier_vide`        |
| `touch`  | Crée un fichier vide                      | `touch nouveau.txt`         |
| `tree`   | Affiche l'arborescence                    | `tree` ou `tree 2`          |

```java
package fr.terminal.emulator;

import javax.swing.*;
import java.awt.*;
import java.awt.event.*;
import java.util.ArrayList;
import java.util.List;

/**
 * Panel principal du terminal.
 * Gère l'affichage de la sortie et la saisie des commandes.
 */
public class TerminalPanel extends JPanel {
    
    private static final Color BG_COLOR = new Color(30, 30, 30);
    private static final Color FG_COLOR = new Color(240, 240, 240);
    private static final Color PROMPT_COLOR = new Color(100, 200, 100);
    private static final Font TERMINAL_FONT = new Font("Monospaced", Font.PLAIN, 14);
    
    private final TerminalOutput outputArea;
    private final TerminalInput inputField;
    private final CommandInterpreter commandInterpreter;
    private final CommandHistory commandHistory;
    
    private String currentPrompt = "user@terminal:~$ ";
    
    /**
     * Crée un nouveau TerminalPanel.
     */
    public TerminalPanel() {
        setLayout(new BorderLayout());
        setBackground(BG_COLOR);
        setForeground(FG_COLOR);
        setFont(TERMINAL_FONT);
        
        // Initialiser le système de fichiers et l'interpréteur
        VirtualFileSystem fs = new VirtualFileSystem();
        this.commandInterpreter = new CommandInterpreter(fs);
        this.commandHistory = new CommandHistory();
        
        // Créer les composants
        this.outputArea = new TerminalOutput();
        this.inputField = new TerminalInput(this::executeCommand, commandHistory);
        
        // Ajouter les composants au panel
        add(new JScrollPane(outputArea), BorderLayout.CENTER);
        add(inputField, BorderLayout.SOUTH);
        
        // Message de bienvenue
        outputArea.println("========================================");
        outputArea.println("  Émulateur de Terminal Unix/Linux");
        outputArea.println("  Tapez 'help' pour la liste des commandes");
        outputArea.println("========================================");
        outputArea.println("");
        
        // Afficher le prompt initial
        displayPrompt();
        inputField.requestFocusInWindow();
    }
    
    /**
     * Affiche le prompt à l'utilisateur.
     */
    public void displayPrompt() {
        String currentDir = commandInterpreter.getCurrentDirectoryPath();
        currentPrompt = String.format("user@terminal:%s$ ", currentDir);
        outputArea.print(currentPrompt, PROMPT_COLOR);
    }
    
    /**
     * Exécute une commande saisie par l'utilisateur.
     * @param command La commande à exécuter
     */
    public void executeCommand(String command) {
        if (command == null || command.trim().isEmpty()) {
            outputArea.println("");
            displayPrompt();
            return;
        }
        
        // Ajouter à l'historique
        commandHistory.add(command);
        
        // Afficher la commande
        outputArea.println(command, Color.WHITE);
        
        // Exécuter la commande
        String result = commandInterpreter.execute(command);
        
        // Afficher le résultat
        if (result != null && !result.isEmpty()) {
            outputArea.println(result);
        }
        
        // Vérifier si on doit quitter
        if (command.trim().equalsIgnoreCase("exit")) {
            System.exit(0);
        }
        
        // Afficher un nouveau prompt
        outputArea.println("");
        displayPrompt();
        
        // Remettre le focus sur le champ de saisie
        inputField.requestFocusInWindow();
    }
    
    /**
     * Donne accès à l'interpréteur de commandes pour les tests.
     * @return L'interpréteur de commandes
     */
    public CommandInterpreter getCommandInterpreter() {
        return commandInterpreter;
    }
    
    /**
     * Donne accès à l'historique des commandes pour les tests.
     * @return L'historique des commandes
     */
    public CommandHistory getCommandHistory() {
        return commandHistory;
    }
}
```

```java
package fr.terminal.emulator;

import javax.swing.*;
import java.awt.*;
import java.awt.event.*;
import java.util.function.Consumer;

/**
 * Champ de saisie pour les commandes du terminal.
 * Gère la saisie utilisateur avec historique et auto-complétion.
 */
public class TerminalInput extends JTextField {
    
    private static final Color BG_COLOR = new Color(50, 50, 50);
    private static final Color FG_COLOR = new Color(240, 240, 240);
    private static final Color CARET_COLOR = new Color(240, 240, 240);
    
    private final Consumer<String> onCommandEntered;
    private final CommandHistory commandHistory;
    
    /**
     * Crée un nouveau champ de saisie.
     * @param onCommandEntered Callback appelé quand une commande est validée
     * @param commandHistory Historique des commandes
     */
    public TerminalInput(Consumer<String> onCommandEntered, CommandHistory commandHistory) {
        this.onCommandEntered = onCommandEntered;
        this.commandHistory = commandHistory;
        
        setBackground(BG_COLOR);
        setForeground(FG_COLOR);
        setCaretColor(CARET_COLOR);
        setFont(new Font("Monospaced", Font.PLAIN, 14));
        setBorder(BorderFactory.createEmptyBorder(5, 10, 5, 10));
        
        // Gestion des événements clavier
        addKeyListener(new TerminalKeyListener());
    }
    
    /**
     * Écouteur des touches pour le terminal.
     */
    private class TerminalKeyListener extends KeyAdapter {
        @Override
        public void keyPressed(KeyEvent e) {
            if (e.getKeyCode() == KeyEvent.VK_ENTER) {
                String command = getText();
                onCommandEntered.accept(command);
                setText("");
                e.consume();
            } else if (e.getKeyCode() == KeyEvent.VK_UP) {
                String previous = commandHistory.getPrevious();
                if (previous != null) {
                    setText(previous);
                }
                e.consume();
            } else if (e.getKeyCode() == KeyEvent.VK_DOWN) {
                String next = commandHistory.getNext();
                if (next != null) {
                    setText(next);
                } else {
                    setText("");
                }
                e.consume();
            } else if (e.getKeyCode() == KeyEvent.VK_TAB) {
                // Auto-complétion (à implémenter)
                e.consume();
            }
        }
    }
}
```

```java
package fr.terminal.emulator;

import javax.swing.*;
import javax.swing.text.*;
import java.awt.*;

/**
 * Zone de texte pour l'affichage de la sortie du terminal.
 * Permet l'affichage avec couleurs et défilement automatique.
 */
public class TerminalOutput extends JTextPane {
    
    private static final Color BG_COLOR = new Color(30, 30, 30);
    private static final Color FG_COLOR = new Color(240, 240, 240);
    private static final Font TERMINAL_FONT = new Font("Monospaced", Font.PLAIN, 14);
    
    private StyledDocument doc;
    private Style defaultStyle;
    
    /**
     * Crée une nouvelle zone de sortie.
     */
    public TerminalOutput() {
        setBackground(BG_COLOR);
        setForeground(FG_COLOR);
        setFont(TERMINAL_FONT);
        setEditable(false);
        setBorder(BorderFactory.createEmptyBorder(10, 10, 0, 10));
        
        doc = getStyledDocument();
        defaultStyle = doc.addStyle("default", null);
        StyleConstants.setForeground(defaultStyle, FG_COLOR);
        StyleConstants.setFontFamily(defaultStyle, "Monospaced");
        StyleConstants.setFontSize(defaultStyle, 14);
    }
    
    /**
     * Affiche du texte avec la couleur par défaut.
     * @param text Le texte à afficher
     */
    public void print(String text) {
        print(text, FG_COLOR);
    }
    
    /**
     * Affiche du texte avec une couleur spécifique.
     * @param text Le texte à afficher
     * @param color La couleur du texte
     */
    public void print(String text, Color color) {
        try {
            Style style = doc.addStyle("color_" + color.getRGB(), null);
            StyleConstants.setForeground(style, color);
            StyleConstants.setFontFamily(style, "Monospaced");
            StyleConstants.setFontSize(style, 14);
            
            int length = doc.getLength();
            doc.insertString(length, text, style);
            
            // Défilement automatique
            setCaretPosition(doc.getLength());
        } catch (BadLocationException e) {
            System.err.println("Erreur lors de l'affichage: " + e.getMessage());
        }
    }
    
    /**
     * Affiche du texte avec la couleur par défaut, suivi d'un saut de ligne.
     * @param text Le texte à afficher
     */
    public void println(String text) {
        print(text + "\n");
    }
    
    /**
     * Affiche du texte avec une couleur spécifique, suivi d'un saut de ligne.
     * @param text Le texte à afficher
     * @param color La couleur du texte
     */
    public void println(String text, Color color) {
        print(text + "\n", color);
    }
    
    /**
     * Efface tout le contenu de la zone de sortie.
     */
    public void clear() {
        try {
            doc.remove(0, doc.getLength());
        } catch (BadLocationException e) {
            System.err.println("Erreur lors de l'effacement: " + e.getMessage());
        }
    }
}
```

#### VirtualFileSystem.java

```java
package fr.terminal.emulator.filesystem;

import java.util.HashMap;
import java.util.Map;

/**
 * Système de fichiers virtuel pour l'émulateur de terminal.
 * Gère la hiérarchie des fichiers et dossiers en mémoire.
 */
public class VirtualFileSystem {
    
    private final VirtualDirectory root;
    private VirtualDirectory current;
    
    /**
     * Crée un nouveau système de fichiers virtuel.
     * Initialise la racine et le répertoire courant.
     */
    public VirtualFileSystem() {
        this.root = new VirtualDirectory("/", null);
        this.current = root;
        
        // Créer quelques répertoires par défaut
        try {
            current.addDirectory(new VirtualDirectory("bin", current));
            current.addDirectory(new VirtualDirectory("etc", current));
            current.addDirectory(new VirtualDirectory("home", current));
            current.addDirectory(new VirtualDirectory("var", current));
            current.addDirectory(new VirtualDirectory("usr", current));
        } catch (FileSystemException e) {
            System.err.println("Erreur lors de l'initialisation: " + e.getMessage());
        }
    }
    
    /**
     * Donne la racine du système de fichiers.
     * @return Le répertoire racine
     */
    public VirtualDirectory getRoot() {
        return root;
    }
    
    /**
     * Donne le répertoire courant.
     * @return Le répertoire courant
     */
    public VirtualDirectory getCurrentDirectory() {
        return current;
    }
    
    /**
     * Définit le répertoire courant.
     * @param directory Le nouveau répertoire courant
     */
    public void setCurrentDirectory(VirtualDirectory directory) {
        if (directory != null) {
            this.current = directory;
        }
    }
    
    /**
     * Résout un chemin (relatif ou absolu) en un VirtualFile ou VirtualDirectory.
     * @param path Le chemin à résoudre
     * @return Le VirtualFile ou VirtualDirectory correspondant
     * @throws FileSystemException Si le chemin n'existe pas
     */
    public VirtualFile resolvePath(String path) throws FileSystemException {
        if (path == null || path.trim().isEmpty()) {
            return current;
        }
        
        String normalizedPath = PathResolver.normalize(path);
        
        if (normalizedPath.equals("/")) {
            return root;
        }
        
        if (normalizedPath.startsWith("/")) {
            // Chemin absolu
            return resolveAbsolutePath(normalizedPath);
        } else {
            // Chemin relatif
            return resolveRelativePath(normalizedPath);
        }
    }
    
    /**
     * Résout un chemin absolu.
     * @param path Le chemin absolu
     * @return Le VirtualFile ou VirtualDirectory
     * @throws FileSystemException Si le chemin n'existe pas
     */
    private VirtualFile resolveAbsolutePath(String path) throws FileSystemException {
        String[] parts = path.split("/");
        VirtualDirectory dir = root;
        
        for (int i = 1; i < parts.length; i++) {
            String part = parts[i];
            if (part.isEmpty() || part.equals(".")) {
                continue;
            }
            
            if (part.equals("..")) {
                if (dir.getParent() != null) {
                    dir = dir.getParent();
                }
                continue;
            }
            
            // Chercher le fichier ou dossier
            VirtualFile file = dir.getFile(part);
            if (file != null) {
                if (i == parts.length - 1) {
                    return file;
                } else {
                    throw new FileSystemException("Chemin invalide: " + part + " est un fichier, pas un dossier");
                }
            }
            
            VirtualDirectory subDir = dir.getDirectory(part);
            if (subDir != null) {
                dir = subDir;
            } else {
                throw new FileSystemException("Dossier non trouvé: " + part);
            }
        }
        
        return dir;
    }
    
    /**
     * Résout un chemin relatif.
     * @param path Le chemin relatif
     * @return Le VirtualFile ou VirtualDirectory
     * @throws FileSystemException Si le chemin n'existe pas
     */
    private VirtualFile resolveRelativePath(String path) throws FileSystemException {
        String[] parts = path.split("/");
        VirtualDirectory dir = current;
        
        for (int i = 0; i < parts.length; i++) {
            String part = parts[i];
            if (part.isEmpty() || part.equals(".")) {
                continue;
            }
            
            if (part.equals("..")) {
                if (dir.getParent() != null) {
                    dir = dir.getParent();
                }
                continue;
            }
            
            // Chercher le fichier ou dossier
            VirtualFile file = dir.getFile(part);
            if (file != null) {
                if (i == parts.length - 1) {
                    return file;
                } else {
                    throw new FileSystemException("Chemin invalide: " + part + " est un fichier, pas un dossier");
                }
            }
            
            VirtualDirectory subDir = dir.getDirectory(part);
            if (subDir != null) {
                dir = subDir;
            } else {
                throw new FileSystemException("Dossier non trouvé: " + part);
            }
        }
        
        return dir;
    }
    
    /**
     * Donne le chemin absolu du répertoire courant.
     * @return Le chemin absolu
     */
    public String getCurrentDirectoryPath() {
        StringBuilder path = new StringBuilder();
        VirtualDirectory dir = current;
        
        java.util.List<String> parts = new java.util.ArrayList<>();
        while (dir != null && !dir.getName().equals("/")) {
            parts.add(0, dir.getName());
            dir = dir.getParent();
        }
        
        if (parts.isEmpty()) {
            return "/";
        }
        
        path.append("/");
        path.append(String.join("/", parts));
        
        return path.toString();
    }
```

```java
package fr.terminal.emulator.utils;

/**
 * Utilitaire pour la résolution des chemins.
 */
public class PathResolver {
    
    /**
     * Normalise un chemin.
     * @param path Le chemin à normaliser
     * @return Le chemin normalisé
     */
    public static String normalize(String path) {
        if (path == null || path.isEmpty()) {
            return ".";
        }
        
        // Remplacer les barres inverses par des barres normales
        String normalized = path.replace('\\', '/');
        
        // Supprimer les doublons de /
        while (normalized.contains("//")) {
            normalized = normalized.replace("//", "/");
        }
        
        // Supprimer le / final si présent (sauf pour la racine)
        if (normalized.length() > 1 && normalized.endsWith("/")) {
            normalized = normalized.substring(0, normalized.length() - 1);
        }
        
        // Gérer le cas de la racine
        if (normalized.equals("/")) {
            return "/";
        }
        
        // Si le chemin ne commence pas par /, c'est un chemin relatif
        if (!normalized.startsWith("/")) {
            return normalized;
        }
        
        return normalized;
    }
    
    /**
     * Vérifie si un chemin est absolu.
     * @param path Le chemin à vérifier
     * @return true si le chemin est absolu
     */
    public static boolean isAbsolute(String path) {
        if (path == null || path.isEmpty()) {
            return false;
        }
        return path.startsWith("/");
    }
    
    /**
     * Donne le nom du fichier/dossier à partir d'un chemin.
     * @param path Le chemin
     * @return Le nom du fichier/dossier
     */
    public static String getBaseName(String path) {
        String normalized = normalize(path);
        
        if (normalized.equals("/")) {
            return "/";
        }
        
        int lastSlash = Math.max(normalized.lastIndexOf('/'), normalized.lastIndexOf('\\'));
        
        if (lastSlash < 0) {
            return normalized;
        }
        
        if (lastSlash == normalized.length() - 1) {
            return "";
        }
        
        return normalized.substring(lastSlash + 1);
    }
    
    /**
     * Donne le répertoire parent à partir d'un chemin.
     * @param path Le chemin
     * @return Le répertoire parent
     */
    public static String getParentPath(String path) {
        String normalized = normalize(path);
        
        if (normalized.equals("/") || normalized.isEmpty()) {
            return "/";
        }
        
        int lastSlash = Math.max(normalized.lastIndexOf('/'), normalized.lastIndexOf('\\'));
        
        if (lastSlash <= 0) {
            return "/";
        }
        
        return normalized.substring(0, lastSlash);
    }
}
```

```java
package fr.terminal.emulator.commands;

import fr.terminal.emulator.filesystem.*;
import fr.terminal.emulator.utils.PathResolver;
import java.util.*;

/**
 * Interpréteur de commandes pour l'émulateur de terminal.
 * Parse et exécute les commandes saisies par l'utilisateur.
 */
public class CommandInterpreter {
    
    private final VirtualFileSystem fileSystem;
    private final Map<String, Command> commands;
    
    /**
     * Crée un nouvel interpréteur de commandes.
     * @param fileSystem Le système de fichiers à utiliser
     */
    public CommandInterpreter(VirtualFileSystem fileSystem) {
        this.fileSystem = fileSystem;
        this.commands = new HashMap<>();
        
        // Enregistrer toutes les commandes
        registerCommands();
    }
    
    /**
     * Enregistre toutes les commandes disponibles.
     */
    private void registerCommands() {
        // Commandes de base
        registerCommand(new HelpCommand(this));
        registerCommand(new ExitCommand());
        registerCommand(new ClearCommand(() -> fileSystem));
        
        // Commandes de navigation
        registerCommand(new PwdCommand(fileSystem));
        registerCommand(new CdCommand(fileSystem));
        registerCommand(new LsCommand(fileSystem));
        registerCommand(new TreeCommand(fileSystem));
        
        // Commandes de manipulation de fichiers
        registerCommand(new CatCommand(fileSystem));
        registerCommand(new TouchCommand(fileSystem));
        registerCommand(new EchoCommand(fileSystem));
        registerCommand(new CpCommand(fileSystem));
        registerCommand(new MvCommand(fileSystem));
        registerCommand(new RmCommand(fileSystem));
        
        // Commandes de manipulation de dossiers
        registerCommand(new MkdirCommand(fileSystem));
        registerCommand(new RmdirCommand(fileSystem));
    }
    
    /**
     * Enregistre une commande.
     * @param command La commande à enregistrer
     */
    public void registerCommand(Command command) {
        if (command != null) {
            commands.put(command.getName().toLowerCase(), command);
        }
    }
    
    /**
     * Exécute une commande.
     * @param commandLine La ligne de commande complète
     * @return Le résultat de l'exécution
     */
    public String execute(String commandLine) {
        if (commandLine == null || commandLine.trim().isEmpty()) {
            return "";
        }
        
        String trimmed = commandLine.trim();
        String[] parts = splitCommandLine(trimmed);
        
        if (parts.length == 0) {
            return "";
        }
        
        String commandName = parts[0].toLowerCase();
        String[] args = new String[parts.length - 1];
        System.arraycopy(parts, 1, args, 0, parts.length - 1);
        
        // Chercher la commande
        Command command = commands.get(commandName);
        
        if (command == null) {
            return "Commande non trouvée: " + commandName + ". Tapez 'help' pour la liste des commandes.";
        }
        
        try {
            return command.execute(args);
        } catch (Exception e) {
            return "Erreur lors de l'exécution de la commande: " + e.getMessage();
        }
    }
    
    /**
     * Donne une commande par son nom.
     * @param name Le nom de la commande
     * @return La commande, ou null si non trouvée
     */
    public Command getCommand(String name) {
        return commands.get(name.toLowerCase());
    }
    
    /**
     * Donne toutes les commandes enregistrées.
     * @return Map des commandes (nom -> commande)
     */
    public Map<String, Command> getCommands() {
        return Collections.unmodifiableMap(commands);
    }
    
    /**
     * Donne le système de fichiers.
     * @return Le système de fichiers
     */
    public VirtualFileSystem getFileSystem() {
        return fileSystem;
    }
    
    /**
     * Donne le chemin du répertoire courant.
     * @return Le chemin absolu du répertoire courant
     */
    public String getCurrentDirectoryPath() {
        return fileSystem.getCurrentDirectoryPath();
    }
    
    /**
     * Divise une ligne de commande en parties.
     * Gère les guillemets et les espaces.
     * @param commandLine La ligne de commande
     * @return Tableau des parties
     */
    private String[] splitCommandLine(String commandLine) {
        List<String> parts = new ArrayList<>();
        StringBuilder current = new StringBuilder();
        boolean inQuotes = false;
        char quoteChar = ' ';
        
        for (int i = 0; i < commandLine.length(); i++) {
            char c = commandLine.charAt(i);
            
            if (c == '\\' && i + 1 < commandLine.length()) {
                // Échappement
                current.append(commandLine.charAt(i + 1));
                i++;
                continue;
            }
            
            if (!inQuotes && (c == '"' || c == '\'')) {
                inQuotes = true;
                quoteChar = c;
                continue;
            }
            
            if (inQuotes && c == quoteChar) {
                inQuotes = false;
                continue;
            }
            
            if (!inQuotes && Character.isWhitespace(c)) {
                if (current.length() > 0) {
                    parts.add(current.toString());
                    current.setLength(0);
                }
                continue;
            }
            
            current.append(c);
        }
        
        if (current.length() > 0) {
            parts.add(current.toString());
        }
        
        return parts.toArray(new String[0]);
    }
}
```

```java
package fr.terminal.emulator.commands;

import fr.terminal.emulator.filesystem.*;
import java.util.List;

/**
 * Commande 'ls' : Liste les fichiers et dossiers.
 */
public class LsCommand implements Command {
    
    private final VirtualFileSystem fileSystem;
    
    /**
     * Crée la commande ls.
     * @param fileSystem Le système de fichiers
     */
    public LsCommand(VirtualFileSystem fileSystem) {
        this.fileSystem = fileSystem;
    }
    
    @Override
    public String getName() {
        return "ls";
    }
    
    @Override
    public String getDescription() {
        return "Liste les fichiers et dossiers";
    }
    
    @Override
    public String getUsage() {
        return "ls [-l] [directory]";
    }
    
    @Override
    public String execute(String[] args) {
        boolean longFormat = false;
        String path = null;
        
        // Parser les arguments
        for (String arg : args) {
            if (arg.equals("-l")) {
                longFormat = true;
            } else if (arg.startsWith("-")) {
                return "Option non reconnue: " + arg;
            } else if (path == null) {
                path = arg;
            } else {
                return "Trop d'arguments";
            }
        }
        
        VirtualDirectory directory;
        
        try {
            if (path != null) {
                VirtualFile target = fileSystem.resolvePath(path);
                if (target instanceof VirtualDirectory) {
                    directory = (VirtualDirectory) target;
                } else {
                    return "Erreur: " + path + " n'est pas un dossier";
                }
            } else {
                directory = fileSystem.getCurrentDirectory();
            }
            
            List<String> contents = directory.listContents();
            
            if (contents.isEmpty()) {
                return "(vide)";
            }
            
            if (longFormat) {
                return listLongFormat(directory);
            } else {
                return String.join("  ", contents);
            }
        } catch (FileSystemException e) {
            return "Erreur: " + e.getMessage();
        }
    }
    
    /**
     * Affiche la liste au format long.
     * @param directory Le dossier à lister
     * @return La chaîne de caractères formatée
     */
    private String listLongFormat(VirtualDirectory directory) {
        StringBuilder sb = new StringBuilder();
        
        // Ajouter les entrées . et ..
        if (directory.getParent() != null) {
            sb.append(String.format("%-10s %-10s %s\n", "drwxr-xr-x", "2", ".."));
        }
        sb.append(String.format("%-10s %-10s %s\n", "drwxr-xr-x", directory.size() + "", "."));
        
        // Lister les sous-dossiers
        for (VirtualDirectory subDir : directory.getDirectories()) {
            sb.append(String.format("%-10s %-10s %s\n", "drwxr-xr-x", "2", subDir.getName()));
        }
        
        // Lister les fichiers
        for (VirtualFile file : directory.getFiles()) {
            sb.append(String.format("%-10s %-10s %s\n", "-rw-r--r--", file.getSize() + "", file.getName()));
        }
        
        return sb.toString().trim();
    }
}
```

```java
public class EchoCommand implements Command {

    /**
     * Crée la commande echo.
     * @param fileSystem Le système de fichiers
     */
    public EchoCommand(VirtualFileSystem fileSystem) {
        this.fileSystem = fileSystem;
    }
    
    @Override
    public String getName() {
        return "echo";
    }
    
    @Override
    public String getDescription() {
        return "Affiche du texte ou écrit dans un fichier";
    }
    
    @Override
    public String getUsage() {
        return "echo [-n] [text] [> file]";
    }
    
    @Override
    public String execute(String[] args) {
        if (args.length == 0) {
            return "";
        }
        
        boolean noNewline = false;
        String outputFile = null;
        StringBuilder text = new StringBuilder();
        
        // Parser les arguments
        for (int i = 0; i < args.length; i++) {
            String arg = args[i];
            
            if (arg.equals("-n")) {
                noNewline = true;
            } else if (arg.equals(">") && i + 1 < args.length) {
                outputFile = args[++i];
            } else if (arg.equals(">>") && i + 1 < args.length) {
                outputFile = args[++i];
                // Mode append (géré plus tard)
            } else {
                if (text.length() > 0) {
                    text.append(" ");
                }
                text.append(arg);
            }
        }
        
        String content = text.toString();
        
        if (outputFile != null) {
            // Écrire dans un fichier
            try {
                String fileName = PathResolver.getBaseName(outputFile);
                String parentPath = PathResolver.getParentPath(outputFile);
                
                VirtualDirectory parentDir;
                if (parentPath.equals("/")) {
                    parentDir = fileSystem.getRoot();
                } else {
                    VirtualFile parent = fileSystem.resolvePath(parentPath);
                    if (parent instanceof VirtualDirectory) {
                        parentDir = (VirtualDirectory) parent;
                    } else {
                        return "Erreur: " + parentPath + " n'est pas un dossier";
                    }
                }
                
                VirtualFile file = parentDir.getFile(fileName);
                
                if (file == null) {
                    file = new VirtualFile(fileName);
                    parentDir.addFile(file);
                }
                
                file.setContent(content);
                return "";
            } catch (FileSystemException e) {
                return "Erreur: " + e.getMessage();
            }
        } else {
            // Afficher à l'écran
            if (noNewline) {
                return content;
            } else {
                return content + "\n";
            }
        }
    }
```

```java
package fr.terminal.emulator.commands;

import fr.terminal.emulator.filesystem.*;
import java.util.List;

/**
 * Commande 'tree' : Affiche l'arborescence des fichiers.
 */
public class TreeCommand implements Command {
    
    private final VirtualFileSystem fileSystem;
    
    /**
     * Crée la commande tree.
     * @param fileSystem Le système de fichiers
     */
    public TreeCommand(VirtualFileSystem fileSystem) {
        this.fileSystem = fileSystem;
    }
    
    @Override
    public String getName() {
        return "tree";
    }
    
    @Override
    public String getDescription() {
        return "Affiche l'arborescence des fichiers et dossiers";
    }
    
    @Override
    public String getUsage() {
        return "tree [depth]";
    }
    
    @Override
    public String execute(String[] args) {
        int maxDepth = Integer.MAX_VALUE;
        
        if (args.length > 0) {
            try {
                maxDepth = Integer.parseInt(args[0]);
            } catch (NumberFormatException e) {
                return "Erreur: " + args[0] + " n'est pas un nombre valide";
            }
        }
        
        VirtualDirectory root = fileSystem.getCurrentDirectory();
        StringBuilder sb = new StringBuilder();
        
        buildTree(root, "", true, 0, maxDepth, sb);
        
        return sb.toString();
    }
    
    /**
     * Construit récursivement l'arborescence.
     * @param directory Le dossier à afficher
     * @param prefix Le préfixe pour l'indentation
     * @param isLast Indique si c'est le dernier élément
     * @param depth La profondeur actuelle
     * @param maxDepth La profondeur maximale
     * @param sb Le StringBuilder pour le résultat
     */
    private void buildTree(VirtualDirectory directory, String prefix, boolean isLast, 
                          int depth, int maxDepth, StringBuilder sb) {
        if (depth > maxDepth) {
            return;
        }
        
        // Afficher le dossier courant
        sb.append(prefix);
        sb.append(isLast ? "└── " : "├── ");
        sb.append(directory.getName());
        sb.append("\n");
        
        // Si on a atteint la profondeur maximale, on s'arrête
        if (depth >= maxDepth) {
            return;
        }
        
        // Préparer le nouveau préfixe
        String newPrefix = prefix + (isLast ? "    " : "│   ");
        
        // Lister tous les éléments (dossiers d'abord, puis fichiers)
        List<String> allItems = directory.listContents();
        
        // Séparer dossiers et fichiers
        java.util.List<String> dirs = new java.util.ArrayList<>();
        java.util.List<String> files = new java.util.ArrayList<>();
        
        for (String item : allItems) {
            if (directory.getDirectory(item) != null) {
                dirs.add(item);
            } else {
                files.add(item);
            }
        }
        
        // Afficher les dossiers
        for (int i = 0; i < dirs.size(); i++) {
            String dirName = dirs.get(i);
            VirtualDirectory subDir = directory.getDirectory(dirName);
            boolean last = (i == dirs.size() - 1 && files.isEmpty()) || 
                          (i == dirs.size() - 1 && !files.isEmpty());
            buildTree(subDir, newPrefix, last, depth + 1, maxDepth, sb);
        }
        
        // Afficher les fichiers
        for (int i = 0; i < files.size(); i++) {
            String fileName = files.get(i);
            sb.append(newPrefix);
            sb.append((dirs.isEmpty() && i == files.size() - 1) || 
                     (!dirs.isEmpty() && i == files.size() - 1) ? "└── " : "├── ");
            sb.append(fileName);
            sb.append("\n");
        }
    }
}
```

```java
package fr.terminal.emulator.utils;

import java.util.ArrayList;
import java.util.List;

/**
 * Gère l'historique des commandes saisies par l'utilisateur.
 */
public class CommandHistory {
    
    private final List<String> history;
    private int currentIndex;
    
    /**
     * Crée un nouvel historique de commandes.
     */
    public CommandHistory() {
        this.history = new ArrayList<>();
        this.currentIndex = -1;
    }
    
    /**
     * Ajoute une commande à l'historique.
     * @param command La commande à ajouter
     */
    public void add(String command) {
        if (command != null && !command.trim().isEmpty()) {
            // Ne pas ajouter de doublons consécutifs
            if (!history.isEmpty() && history.get(history.size() - 1).equals(command)) {
                return;
            }
            history.add(command);
            currentIndex = history.size();
        }
    }
    
    /**
     * Donne la commande précédente dans l'historique.
     * @return La commande précédente, ou null si on est au début
     */
    public String getPrevious() {
        if (history.isEmpty()) {
            return null;
        }
        
        if (currentIndex > 0) {
            currentIndex--;
            return history.get(currentIndex);
        } else {
            return null;
        }
    }
    
    /**
     * Donne la commande suivante dans l'historique.
     * @return La commande suivante, ou null si on est à la fin
     */
    public String getNext() {
        if (history.isEmpty()) {
            return null;
        }
        
        if (currentIndex < history.size() - 1) {
            currentIndex++;
            return history.get(currentIndex);
        } else {
            currentIndex = history.size();
            return null;
        }
    }
    
    /**
     * Réinitialise l'index pour une nouvelle saisie.
     */
    public void resetIndex() {
        currentIndex = history.size();
    }
    
    /**
     * Donne toute l'historique.
     * @return Liste de toutes les commandes
     */
    public List<String> getHistory() {
        return new ArrayList<>(history);
    }
    
    /**
     * Efface l'historique.
     */
    public void clear() {
        history.clear();
        currentIndex = -1;
    }
    
    /**
     * Donne le nombre de commandes dans l'historique.
     * @return Le nombre de commandes
     */
    public int size() {
        return history.size();
    }
}
```

```java
package fr.terminal.emulator.filesystem;

import org.junit.jupiter.api.*;
import static org.junit.jupiter.api.Assertions.*;

/**
 * Tests unitaires pour le système de fichiers virtuel.
 */
public class VirtualFileSystemTest {
    
    private VirtualFileSystem fs;
    
    @BeforeEach
    public void setUp() {
        fs = new VirtualFileSystem();
    }
    
    @Test
    public void testRootDirectory() {
        VirtualDirectory root = fs.getRoot();
        assertNotNull(root);
        assertEquals("/", root.getName());
        assertNull(root.getParent());
    }
    
    @Test
    public void testCurrentDirectoryInitiallyRoot() {
        VirtualDirectory current = fs.getCurrentDirectory();
        assertEquals(fs.getRoot(), current);
    }
    
    @Test
    public void testResolveRootPath() throws FileSystemException {
        VirtualFile root = fs.resolvePath("/");
        assertEquals(fs.getRoot(), root);
    }
    
    @Test
    public void testResolveCurrentDirectory() throws FileSystemException {
        VirtualFile current = fs.resolvePath(".");
        assertEquals(fs.getCurrentDirectory(), current);
    }
    
    @Test
    public void testResolveParentDirectory() throws FileSystemException {
        VirtualDirectory current = fs.getCurrentDirectory();
        fs.setCurrentDirectory(current.getDirectory("home"));
        
        VirtualFile parent = fs.resolvePath("..");
        assertEquals(fs.getRoot(), parent);
    }
    
    @Test
    public void testCreateAndResolveFile() throws FileSystemException {
        VirtualDirectory home = fs.getRoot().getDirectory("home");
        VirtualFile file = new VirtualFile("test.txt");
        home.addFile(file);
        
        VirtualFile resolved = fs.resolvePath("/home/test.txt");
        assertEquals(file, resolved);
    }
    
    @Test
    public void testCreateAndResolveDirectory() throws FileSystemException {
        VirtualDirectory home = fs.getRoot().getDirectory("home");
        VirtualDirectory newDir = new VirtualDirectory("projects", home);
        home.addDirectory(newDir);
        
        VirtualFile resolved = fs.resolvePath("/home/projects");
        assertEquals(newDir, resolved);
    }
    
    @Test
    public void testRelativePathResolution() throws FileSystemException {
        VirtualDirectory home = fs.getRoot().getDirectory("home");
        fs.setCurrentDirectory(home);
        
        VirtualDirectory projects = new VirtualDirectory("projects", home);
        home.addDirectory(projects);
        
        VirtualFile resolved = fs.resolvePath("projects");
        assertEquals(projects, resolved);
    }
    
    @Test
    public void testInvalidPath() {
        assertThrows(FileSystemException.class, () -> {
            fs.resolvePath("/nonexistent");
        });
    }
    
    @Test
    public void testCurrentDirectoryPath() {
        assertEquals("/", fs.getCurrentDirectoryPath());
        
        VirtualDirectory home = fs.getRoot().getDirectory("home");
        fs.setCurrentDirectory(home);
        assertEquals("/home", fs.getCurrentDirectoryPath());
    }
}
```

```java
import org.junit.jupiter.api.*;

/**

public class CommandInterpreterTest {

    @BeforeEach
    public void setUp() {

        interpreter = new CommandInterpreter(fs);
    }

    }

        String result = interpreter.execute("pwd");

    @Test

        String result = interpreter.execute("ls");
        assertTrue(result.contains("bin"));
        assertTrue(result.contains("etc"));
        assertTrue(result.contains("home"));

    @Test
    public void testTouchAndCatCommand() {

        String result = interpreter.execute("cat test.txt");
        assertEquals("Hello", result);
    }
    
    @Test
    public void testMkdirCommand() {
        interpreter.execute("mkdir testdir");
        String result = interpreter.execute("ls");
        assertTrue(result.contains("testdir"));
    }
    
    @Test
    public void testCdToNonexistent() {
        String result = interpreter.execute("cd /nonexistent");
        assertTrue(result.contains("Erreur"));
    }
    
    @Test
    public void testUnknownCommand() {
        String result = interpreter.execute("unknowncommand");
        assertTrue(result.contains("Commande non trouvée"));
    }
    
    @Test
    public void testHelpCommand() {
        String result = interpreter.execute("help");
        assertTrue(result.contains("Commandes disponibles"));
        assertTrue(result.contains("help"));
        assertTrue(result.contains("pwd"));
    }
    
    @Test
    public void testHelpSpecificCommand() {
        String result = interpreter.execute("help pwd");
        assertTrue(result.contains("pwd"));
        assertTrue(result.contains("Description"));
    }
    
    @Test
    public void testCpCommand() {
        interpreter.execute("touch source.txt");
        interpreter.execute("echo Content > source.txt");
        interpreter.execute("cp source.txt dest.txt");
        String result = interpreter.execute("cat dest.txt");
        assertEquals("Content", result);
    }
    
    @Test
    public void testMvCommand() {
        interpreter.execute("touch old.txt");
        interpreter.execute("mv old.txt new.txt");
        String result = interpreter.execute("ls");
        assertTrue(result.contains("new.txt"));
        assertFalse(result.contains("old.txt"));
    }

        interpreter.execute("rm todelete.txt");

    }

    public void testMkdirAndRmdir() {
```

## 📚 Documentation Technique

### Concepts Clés

#### 1. **Système de Fichiers Virtuel**

Le système de fichiers est entièrement implémenté en mémoire avec :

- `VirtualFileSystem` : Gère la racine et le répertoire courant
- `VirtualDirectory` : Représente un dossier avec ses sous-éléments
- `VirtualFile` : Représente un fichier avec son contenu
- `FileSystemException` : Pour la gestion des erreurs

**Caractéristiques** :

- Hiérarchie arborescente classique
- Chemins absolus (`/home/user`) et relatifs (`../parent`)
- Gestion des noms de fichiers/dossiers uniques dans un même répertoire

#### 2. **Interpréteur de Commandes**

L'interpréteur (`CommandInterpreter`) :

- Parse la ligne de commande en tokens
- Résout les chemins de fichiers
- Trouve et exécute la commande appropriée
- Gère les erreurs et retourne les résultats

**Architecture des commandes** :

- Chaque commande implémente l'interface `Command`
- Méthodes requises : `getName()`, `getDescription()`, `getUsage()`, `execute(String[])`
- Les commandes reçoivent le système de fichiers via injection de dépendance

#### 3. **Interface Graphique Swing**

L'interface utilise :

- `TerminalFrame` : Fenêtre principale
- `TerminalPanel` : Panel principal avec entrée/sortie
- `TerminalOutput` : Zone de texte pour l'affichage (avec couleurs)
- `TerminalInput` : Champ de saisie pour les commandes

**Fonctionnalités UI** :

- Thème sombre par défaut
- Police monospace
- Historique des commandes (↑/↓)
- Défilement automatique
- Prompt personnalisable

#### 4. **Gestion des Chemins**

`PathResolver` fournit des utilitaires pour :

- Normalisation des chemins (`//` → `/`, `\` → `/`)
- Extraction du nom de base (`/home/user/file.txt` → `file.txt`)
- Extraction du parent (`/home/user/file.txt` → `/home/user`)
- Détection des chemins absolus vs relatifs

### 1. Visualisation Desktop (Priorité 1)

Pour ajouter une visualisation Desktop comme demandé :

```java
// Nouvelle classe DesktopView.java
package fr.terminal.emulator;

import javax.swing.*;
import java.awt.*;
import fr.terminal.emulator.filesystem.*;

/**
 * Vue Desktop pour visualiser les fichiers sous forme d'icônes.
 */
public class DesktopView extends JPanel {
    
    private final VirtualFileSystem fileSystem;
    private final int iconSize = 64;
    private final int margin = 20;
    
    public DesktopView(VirtualFileSystem fileSystem) {
        this.fileSystem = fileSystem;
        setLayout(new FlowLayout(FlowLayout.LEFT, margin, margin));
        setBackground(new Color(40, 40, 40));
        refresh();
    }
    
    /**
     * Rafraîchit l'affichage avec le contenu du répertoire courant.
     */
    public void refresh() {
        removeAll();
        
        VirtualDirectory current = fileSystem.getCurrentDirectory();
        
        // Ajouter l'icône "Parent"
        if (current.getParent() != null) {
            add(createIconPanel("../", "dossier_parent.png", () -> {
                fileSystem.setCurrentDirectory(current.getParent());
                refresh();
            }));
        }
        
        // Ajouter les icônes des sous-dossiers
        for (VirtualDirectory dir : current.getDirectories()) {
            add(createIconPanel(dir.getName() + "/", "dossier.png", () -> {
                fileSystem.setCurrentDirectory(dir);
                refresh();
            }));
        }
        
        // Ajouter les icônes des fichiers
        for (VirtualFile file : current.getFiles()) {
            add(createIconPanel(file.getName(), "fichier.png", () -> {
                // Ouvrir le fichier (à implémenter)
                JOptionPane.showMessageDialog(this, 
                    file.getContent(), file.getName(), 
                    JOptionPane.INFORMATION_MESSAGE);
            }));
        }
        
        revalidate();
        repaint();
    }
    
    private JPanel createIconPanel(String label, String iconName, Runnable action) {
        JPanel panel = new JPanel();
        panel.setLayout(new BorderLayout());
        panel.setPreferredSize(new Dimension(iconSize + 20, iconSize + 40));
        panel.setBackground(new Color(40, 40, 40));
        
        // Icône (simplifiée - en vrai on utiliserait ImageIO)
        JLabel iconLabel = new JLabel("📁", SwingConstants.CENTER);
        iconLabel.setFont(new Font("Segoe UI Emoji", Font.PLAIN, 40));
        iconLabel.setPreferredSize(new Dimension(iconSize, iconSize));
        
        // Label du nom
        JLabel nameLabel = new JLabel(label, SwingConstants.CENTER);
        nameLabel.setForeground(Color.WHITE);
        nameLabel.setFont(new Font("SansSerif", Font.PLAIN, 12));
        
        panel.add(iconLabel, BorderLayout.CENTER);
        panel.add(nameLabel, BorderLayout.SOUTH);
        
        // Gestion du clic
        panel.setCursor(new Cursor(Cursor.HAND_CURSOR));
        panel.addMouseListener(new java.awt.event.MouseAdapter() {
            @Override
            public void mouseClicked(java.awt.event.MouseEvent e) {
                if (e.getClickCount() == 2) {
                    action.run();
                }
            }
        });
        
        return panel;
    }
}
```

**Intégration avec la fenêtre principale** :

```java
// Modifier TerminalFrame.java
public class TerminalFrame extends JFrame {
    private TerminalPanel terminalPanel;
    private DesktopView desktopView;
    private JTabbedPane tabbedPane;
    
    public TerminalFrame() {
        // ... configuration de base
        
        tabbedPane = new JTabbedPane();
        
        // Onglet Terminal
        terminalPanel = new TerminalPanel();
        tabbedPane.addTab("Terminal", terminalPanel);
        
        // Onglet Desktop
        VirtualFileSystem fs = terminalPanel.getCommandInterpreter().getFileSystem();
        desktopView = new DesktopView(fs);
        tabbedPane.addTab("Desktop", new JScrollPane(desktopView));
        
        add(tabbedPane, BorderLayout.CENTER);
        
        // Synchroniser les changements de répertoire
        terminalPanel.getCommandInterpreter().getFileSystem()
            .addChangeListener(() -> desktopView.refresh());
    }
}
```

```java
package fr.terminal.emulator;

import javax.swing.*;
import java.awt.*;
import java.awt.event.*;
import java.util.function.Consumer;

/**
 * Champ de saisie pour les commandes du terminal.
 * Gère la saisie utilisateur avec historique et auto-complétion.
 */
public class TerminalInput extends JTextField {
    
    private static final Color BG_COLOR = new Color(50, 50, 50);
    private static final Color FG_COLOR = new Color(240, 240, 240);
    private static final Color CARET_COLOR = new Color(240, 240, 240);
    
    private final Consumer<String> onCommandEntered;
    private final CommandHistory commandHistory;
    
    /**
     * Crée un nouveau champ de saisie.
     * @param onCommandEntered Callback appelé quand une commande est validée
     * @param commandHistory Historique des commandes
     */
    public TerminalInput(Consumer<String> onCommandEntered, CommandHistory commandHistory) {
        this.onCommandEntered = onCommandEntered;
        this.commandHistory = commandHistory;
        
        setBackground(BG_COLOR);
        setForeground(FG_COLOR);
        setCaretColor(CARET_COLOR);
        setFont(new Font("Monospaced", Font.PLAIN, 14));
        setBorder(BorderFactory.createEmptyBorder(5, 10, 5, 10));
        
        // Gestion des événements clavier
        addKeyListener(new TerminalKeyListener());
    }
    
    /**
     * Écouteur des touches pour le terminal.
     */
    private class TerminalKeyListener extends KeyAdapter {
        @Override
        public void keyPressed(KeyEvent e) {
            if (e.getKeyCode() == KeyEvent.VK_ENTER) {
                String command = getText();
                onCommandEntered.accept(command);
                setText("");
                e.consume();
            } else if (e.getKeyCode() == KeyEvent.VK_UP) {
                String previous = commandHistory.getPrevious();
                if (previous != null) {
                    setText(previous);
                }
                e.consume();
            } else if (e.getKeyCode() == KeyEvent.VK_DOWN) {
                String next = commandHistory.getNext();
                if (next != null) {
                    setText(next);
                } else {
                    setText("");
                }
                e.consume();
            } else if (e.getKeyCode() == KeyEvent.VK_TAB) {
                // Auto-complétion (à implémenter)
                e.consume();
            }
        }
    }
}
```

```java
package fr.terminal.emulator.filesystem;

import java.util.HashMap;
import java.util.Map;

/**
 * Système de fichiers virtuel pour l'émulateur de terminal.
 * Gère la hiérarchie des fichiers et dossiers en mémoire.
 */
public class VirtualFileSystem {
    
    private final VirtualDirectory root;
    private VirtualDirectory current;
    
    /**
     * Crée un nouveau système de fichiers virtuel.
     * Initialise la racine et le répertoire courant.
     */
    public VirtualFileSystem() {
        this.root = new VirtualDirectory("/", null);
        this.current = root;
        
        // Créer quelques répertoires par défaut
        try {
            current.addDirectory(new VirtualDirectory("bin", current));
            current.addDirectory(new VirtualDirectory("etc", current));
            current.addDirectory(new VirtualDirectory("home", current));
            current.addDirectory(new VirtualDirectory("var", current));
            current.addDirectory(new VirtualDirectory("usr", current));
        } catch (FileSystemException e) {
            System.err.println("Erreur lors de l'initialisation: " + e.getMessage());
        }
    }
    
    /**
     * Donne la racine du système de fichiers.
     * @return Le répertoire racine
     */
    public VirtualDirectory getRoot() {
        return root;
    }
    
    /**
     * Donne le répertoire courant.
     * @return Le répertoire courant
     */
    public VirtualDirectory getCurrentDirectory() {
        return current;
    }
    
    /**
     * Définit le répertoire courant.
     * @param directory Le nouveau répertoire courant
     */
    public void setCurrentDirectory(VirtualDirectory directory) {
        if (directory != null) {
            this.current = directory;
        }
    }
    
    /**
     * Résout un chemin (relatif ou absolu) en un VirtualFile ou VirtualDirectory.
     * @param path Le chemin à résoudre
     * @return Le VirtualFile ou VirtualDirectory correspondant
     * @throws FileSystemException Si le chemin n'existe pas
     */
    public VirtualFile resolvePath(String path) throws FileSystemException {
        if (path == null || path.trim().isEmpty()) {
            return current;
        }
        
        String normalizedPath = PathResolver.normalize(path);
        
        if (normalizedPath.equals("/")) {
            return root;
        }
        
        if (normalizedPath.startsWith("/")) {
            // Chemin absolu
            return resolveAbsolutePath(normalizedPath);
        } else {
            // Chemin relatif
            return resolveRelativePath(normalizedPath);
        }
    }
    
    /**
     * Résout un chemin absolu.
     * @param path Le chemin absolu
     * @return Le VirtualFile ou VirtualDirectory
     * @throws FileSystemException Si le chemin n'existe pas
     */
    private VirtualFile resolveAbsolutePath(String path) throws FileSystemException {
        String[] parts = path.split("/");
        VirtualDirectory dir = root;
        
        for (int i = 1; i < parts.length; i++) {
            String part = parts[i];
            if (part.isEmpty() || part.equals(".")) {
                continue;
            }
            
            if (part.equals("..")) {
                if (dir.getParent() != null) {
                    dir = dir.getParent();
                }
                continue;
            }
            
            // Chercher le fichier ou dossier
            VirtualFile file = dir.getFile(part);
            if (file != null) {
                if (i == parts.length - 1) {
                    return file;
                } else {
                    throw new FileSystemException("Chemin invalide: " + part + " est un fichier, pas un dossier");
                }
            }
            
            VirtualDirectory subDir = dir.getDirectory(part);
            if (subDir != null) {
                dir = subDir;
            } else {
                throw new FileSystemException("Dossier non trouvé: " + part);
            }
        }
        
        return dir;
    }
    
    /**
     * Résout un chemin relatif.
     * @param path Le chemin relatif
     * @return Le VirtualFile ou VirtualDirectory
     * @throws FileSystemException Si le chemin n'existe pas
     */
    private VirtualFile resolveRelativePath(String path) throws FileSystemException {
        String[] parts = path.split("/");
        VirtualDirectory dir = current;
        
        for (int i = 0; i < parts.length; i++) {
            String part = parts[i];
            if (part.isEmpty() || part.equals(".")) {
                continue;
            }
            
            if (part.equals("..")) {
                if (dir.getParent() != null) {
                    dir = dir.getParent();
                }
                continue;
            }
            
            // Chercher le fichier ou dossier
            VirtualFile file = dir.getFile(part);
            if (file != null) {
                if (i == parts.length - 1) {
                    return file;
                } else {
                    throw new FileSystemException("Chemin invalide: " + part + " est un fichier, pas un dossier");
                }
            }
            
            VirtualDirectory subDir = dir.getDirectory(part);
            if (subDir != null) {
                dir = subDir;
            } else {
                throw new FileSystemException("Dossier non trouvé: " + part);
            }
        }
        
        return dir;
    }
    
    /**
     * Donne le chemin absolu du répertoire courant.
     * @return Le chemin absolu
     */
    public String getCurrentDirectoryPath() {
        StringBuilder path = new StringBuilder();
        VirtualDirectory dir = current;
        
        java.util.List<String> parts = new java.util.ArrayList<>();
        while (dir != null && !dir.getName().equals("/")) {
            parts.add(0, dir.getName());
            dir = dir.getParent();
        }
        
        if (parts.isEmpty()) {
            return "/";
        }
        
        path.append("/");
        path.append(String.join("/", parts));
        
        return path.toString();
    }
}
```

```java
public class VirtualDirectory extends VirtualFile {

    private final Map<String, VirtualDirectory> directories;
    private final VirtualDirectory parent;
    
    /**
     * Crée un nouveau dossier.
     * @param name Le nom du dossier
     * @param parent Le dossier parent (null pour la racine)
     */
    public VirtualDirectory(String name, VirtualDirectory parent) {
        super(name, "", System.currentTimeMillis());
        this.files = new HashMap<>();
        this.directories = new HashMap<>();
        this.parent = parent;
    }
    
    /**
     * Donne le dossier parent.
     * @return Le dossier parent, ou null si c'est la racine
     */
    public VirtualDirectory getParent() {
        return parent;
    }
    
    /**
     * Ajoute un fichier au dossier.
     * @param file Le fichier à ajouter
     * @throws FileSystemException Si un fichier ou dossier avec le même nom existe déjà
     */
    public void addFile(VirtualFile file) throws FileSystemException {
        if (file == null) {
            throw new IllegalArgumentException("Le fichier ne peut pas être null");
        }
        
        String name = file.getName();
        if (files.containsKey(name) || directories.containsKey(name)) {
            throw new FileSystemException("Un fichier ou dossier avec le nom '" + name + "' existe déjà");
        }
        
        files.put(name, file);
    }
    
    /**
     * Ajoute un sous-dossier au dossier.
     * @param directory Le sous-dossier à ajouter
     * @throws FileSystemException Si un fichier ou dossier avec le même nom existe déjà
     */
    public void addDirectory(VirtualDirectory directory) throws FileSystemException {
        if (directory == null) {
            throw new IllegalArgumentException("Le dossier ne peut pas être null");
        }
        
        String name = directory.getName();
        if (files.containsKey(name) || directories.containsKey(name)) {
            throw new FileSystemException("Un fichier ou dossier avec le nom '" + name + "' existe déjà");
        }
        
        directories.put(name, directory);
    }
    
    /**
     * Donne un fichier par son nom.
     * @param name Le nom du fichier
     * @return Le fichier, ou null s'il n'existe pas
     */
    public VirtualFile getFile(String name) {
        return files.get(name);
    }
    
    /**
     * Donne un sous-dossier par son nom.
     * @param name Le nom du sous-dossier
     * @return Le sous-dossier, ou null s'il n'existe pas
     */
    public VirtualDirectory getDirectory(String name) {
        return directories.get(name);
    }
    
    /**
     * Supprime un fichier.
     * @param name Le nom du fichier à supprimer
     * @return true si le fichier a été supprimé, false sinon
     */
    public boolean removeFile(String name) {
        return files.remove(name) != null;
    }
    
    /**
     * Supprime un sous-dossier.
     * @param name Le nom du sous-dossier à supprimer
     * @return true si le sous-dossier a été supprimé, false sinon
     */
    public boolean removeDirectory(String name) {
        return directories.remove(name) != null;
    }
    
    /**
     * Donne la liste de tous les noms de fichiers et dossiers.
     * @return Liste des noms
     */
    public List<String> listContents() {
        List<String> contents = new ArrayList<>();
        contents.addAll(files.keySet());
        contents.addAll(directories.keySet());
        contents.sort(String::compareTo);
        return contents;
    }

     * Donne la liste des fichiers.
     * @return Liste des fichiers

     * Donne la liste des sous-dossiers.
     * @return Liste des sous-dossiers
     */

    public int size() {

    }
```

```java
 */
public class PathResolver {
    
    /**
     * Normalise un chemin.
     * @param path Le chemin à normaliser
     * @return Le chemin normalisé
     */
    public static String normalize(String path) {
        if (path == null || path.isEmpty()) {
            return ".";
        }
        
        // Remplacer les barres inverses par des barres normales
        String normalized = path.replace('\\', '/');
        
        // Supprimer les doublons de /
        while (normalized.contains("//")) {
            normalized = normalized.replace("//", "/");
        }
        
        // Supprimer le / final si présent (sauf pour la racine)
        if (normalized.length() > 1 && normalized.endsWith("/")) {
            normalized = normalized.substring(0, normalized.length() - 1);
        }
        
        // Gérer le cas de la racine
        if (normalized.equals("/")) {
            return "/";
        }
        
        // Si le chemin ne commence pas par /, c'est un chemin relatif
        if (!normalized.startsWith("/")) {
            return normalized;
        }
        
        return normalized;
    }
    
    /**
     * Vérifie si un chemin est absolu.
     * @param path Le chemin à vérifier
     * @return true si le chemin est absolu
     */
    public static boolean isAbsolute(String path) {
        if (path == null || path.isEmpty()) {
            return false;
        }
        return path.startsWith("/");
    }
    
    /**
     * Donne le nom du fichier/dossier à partir d'un chemin.
     * @param path Le chemin
     * @return Le nom du fichier/dossier
     */
    public static String getBaseName(String path) {
        String normalized = normalize(path);
        
        if (normalized.equals("/")) {
            return "/";
        }
        
        int lastSlash = Math.max(normalized.lastIndexOf('/'), normalized.lastIndexOf('\\'));
        
        if (lastSlash < 0) {
            return normalized;
        }
        
        if (lastSlash == normalized.length() - 1) {
            return "";
        }
        
        return normalized.substring(lastSlash + 1);
    }
    
    /**
     * Donne le répertoire parent à partir d'un chemin.
     * @param path Le chemin
     * @return Le répertoire parent
     */
    public static String getParentPath(String path) {
        String normalized = normalize(path);
        
        if (normalized.equals("/") || normalized.isEmpty()) {
            return "/";
        }
        
        int lastSlash = Math.max(normalized.lastIndexOf('/'), normalized.lastIndexOf('\\'));
        
        if (lastSlash <= 0) {
            return "/";
        }
        
        return normalized.substring(0, lastSlash);
    }
}
```

#### CommandInterpreter.java

```java
package fr.terminal.emulator.commands;

import fr.terminal.emulator.filesystem.*;
import fr.terminal.emulator.utils.PathResolver;
import java.util.*;

/**
 * Interpréteur de commandes pour l'émulateur de terminal.
 * Parse et exécute les commandes saisies par l'utilisateur.
 */
public class CommandInterpreter {
    
    private final VirtualFileSystem fileSystem;
    private final Map<String, Command> commands;
    
    /**
     * Crée un nouvel interpréteur de commandes.
     * @param fileSystem Le système de fichiers à utiliser
     */
    public CommandInterpreter(VirtualFileSystem fileSystem) {
        this.fileSystem = fileSystem;
        this.commands = new HashMap<>();
        
        // Enregistrer toutes les commandes
        registerCommands();
    }
    
    /**
     * Enregistre toutes les commandes disponibles.
     */
    private void registerCommands() {
        // Commandes de base
        registerCommand(new HelpCommand(this));
        registerCommand(new ExitCommand());
        registerCommand(new ClearCommand(() -> fileSystem));
        
        // Commandes de navigation
        registerCommand(new PwdCommand(fileSystem));
        registerCommand(new CdCommand(fileSystem));
        registerCommand(new LsCommand(fileSystem));
        registerCommand(new TreeCommand(fileSystem));
        
        // Commandes de manipulation de fichiers
        registerCommand(new CatCommand(fileSystem));
        registerCommand(new TouchCommand(fileSystem));
        registerCommand(new EchoCommand(fileSystem));
        registerCommand(new CpCommand(fileSystem));
        registerCommand(new MvCommand(fileSystem));
        registerCommand(new RmCommand(fileSystem));
        
        // Commandes de manipulation de dossiers
        registerCommand(new MkdirCommand(fileSystem));
        registerCommand(new RmdirCommand(fileSystem));
    }
    
    /**
     * Enregistre une commande.
     * @param command La commande à enregistrer
     */
    public void registerCommand(Command command) {
        if (command != null) {
            commands.put(command.getName().toLowerCase(), command);
        }
    }
    
    /**
     * Exécute une commande.
     * @param commandLine La ligne de commande complète
     * @return Le résultat de l'exécution
     */
    public String execute(String commandLine) {
        if (commandLine == null || commandLine.trim().isEmpty()) {
            return "";
        }
        
        String trimmed = commandLine.trim();
        String[] parts = splitCommandLine(trimmed);
        
        if (parts.length == 0) {
            return "";
        }
        
        String commandName = parts[0].toLowerCase();
        String[] args = new String[parts.length - 1];
        System.arraycopy(parts, 1, args, 0, parts.length - 1);
        
        // Chercher la commande
        Command command = commands.get(commandName);
        
        if (command == null) {
            return "Commande non trouvée: " + commandName + ". Tapez 'help' pour la liste des commandes.";
        }
        
        try {
            return command.execute(args);
        } catch (Exception e) {
            return "Erreur lors de l'exécution de la commande: " + e.getMessage();
        }
    }
    
    /**
     * Donne une commande par son nom.
     * @param name Le nom de la commande
     * @return La commande, ou null si non trouvée
     */
    public Command getCommand(String name) {
        return commands.get(name.toLowerCase());
    }
    
    /**
     * Donne toutes les commandes enregistrées.
     * @return Map des commandes (nom -> commande)
     */
    public Map<String, Command> getCommands() {
        return Collections.unmodifiableMap(commands);
    }
    
    /**
     * Donne le système de fichiers.
     * @return Le système de fichiers
     */
    public VirtualFileSystem getFileSystem() {
        return fileSystem;
    }
    
    /**
     * Donne le chemin du répertoire courant.
     * @return Le chemin absolu du répertoire courant
     */
    public String getCurrentDirectoryPath() {
        return fileSystem.getCurrentDirectoryPath();
    }
    
    /**
     * Divise une ligne de commande en parties.
     * Gère les guillemets et les espaces.
     * @param commandLine La ligne de commande
     * @return Tableau des parties
     */
    private String[] splitCommandLine(String commandLine) {
        List<String> parts = new ArrayList<>();
        StringBuilder current = new StringBuilder();
        boolean inQuotes = false;
        char quoteChar = ' ';
        
        for (int i = 0; i < commandLine.length(); i++) {
            char c = commandLine.charAt(i);
            
            if (c == '\\' && i + 1 < commandLine.length()) {
                // Échappement
                current.append(commandLine.charAt(i + 1));
                i++;
                continue;
            }
            
            if (!inQuotes && (c == '"' || c == '\'')) {
                inQuotes = true;
                quoteChar = c;
                continue;
            }
            
            if (inQuotes && c == quoteChar) {
                inQuotes = false;
                continue;
            }
            
            if (!inQuotes && Character.isWhitespace(c)) {
                if (current.length() > 0) {
                    parts.add(current.toString());
                    current.setLength(0);
                }
                continue;
            }
            
            current.append(c);
        }
        
        if (current.length() > 0) {
            parts.add(current.toString());
        }
        
        return parts.toArray(new String[0]);
    }
}
```

```java
package fr.terminal.emulator.commands;

import fr.terminal.emulator.filesystem.*;
import java.util.List;

/**
 * Commande 'ls' : Liste les fichiers et dossiers.
 */
public class LsCommand implements Command {
    
    private final VirtualFileSystem fileSystem;
    
    /**
     * Crée la commande ls.
     * @param fileSystem Le système de fichiers
     */
    public LsCommand(VirtualFileSystem fileSystem) {
        this.fileSystem = fileSystem;
    }
    
    @Override
    public String getName() {
        return "ls";
    }
    
    @Override
    public String getDescription() {
        return "Liste les fichiers et dossiers";
    }
    
    @Override
    public String getUsage() {
        return "ls [-l] [directory]";
    }
    
    @Override
    public String execute(String[] args) {
        boolean longFormat = false;
        String path = null;
        
        // Parser les arguments
        for (String arg : args) {
            if (arg.equals("-l")) {
                longFormat = true;
            } else if (arg.startsWith("-")) {
                return "Option non reconnue: " + arg;
            } else if (path == null) {
                path = arg;
            } else {
                return "Trop d'arguments";
            }
        }
        
        VirtualDirectory directory;
        
        try {
            if (path != null) {
                VirtualFile target = fileSystem.resolvePath(path);
                if (target instanceof VirtualDirectory) {
                    directory = (VirtualDirectory) target;
                } else {
                    return "Erreur: " + path + " n'est pas un dossier";
                }
            } else {
                directory = fileSystem.getCurrentDirectory();
            }
            
            List<String> contents = directory.listContents();
            
            if (contents.isEmpty()) {
                return "(vide)";
            }
            
            if (longFormat) {
                return listLongFormat(directory);
            } else {
                return String.join("  ", contents);
            }
        } catch (FileSystemException e) {
            return "Erreur: " + e.getMessage();
        }
    }
    
    /**
     * Affiche la liste au format long.
     * @param directory Le dossier à lister
     * @return La chaîne de caractères formatée
     */
    private String listLongFormat(VirtualDirectory directory) {
        StringBuilder sb = new StringBuilder();
        
        // Ajouter les entrées . et ..
        if (directory.getParent() != null) {
            sb.append(String.format("%-10s %-10s %s\n", "drwxr-xr-x", "2", ".."));
        }
        sb.append(String.format("%-10s %-10s %s\n", "drwxr-xr-x", directory.size() + "", "."));
        
        // Lister les sous-dossiers
        for (VirtualDirectory subDir : directory.getDirectories()) {
            sb.append(String.format("%-10s %-10s %s\n", "drwxr-xr-x", "2", subDir.getName()));
        }
        
        // Lister les fichiers
        for (VirtualFile file : directory.getFiles()) {
            sb.append(String.format("%-10s %-10s %s\n", "-rw-r--r--", file.getSize() + "", file.getName()));
        }
        
        return sb.toString().trim();
    }
}
```

```java
package fr.terminal.emulator.commands;

import fr.terminal.emulator.filesystem.*;
import fr.terminal.emulator.utils.PathResolver;
import java.util.Arrays;

/**
 * Commande 'echo' : Affiche du texte ou écrit dans un fichier.
 */
public class EchoCommand implements Command {
    
    private final VirtualFileSystem fileSystem;
    
    /**
     * Crée la commande echo.
     * @param fileSystem Le système de fichiers
     */
    public EchoCommand(VirtualFileSystem fileSystem) {
        this.fileSystem = fileSystem;

    public String getDescription() {

    @Override
    public String getUsage() {
        return "echo [-n] [text] [> file]";
    }
    
    @Override
    public String execute(String[] args) {
        if (args.length == 0) {
            return "";
        }
        
        boolean noNewline = false;
        String outputFile = null;
        StringBuilder text = new StringBuilder();
        
        // Parser les arguments
        for (int i = 0; i < args.length; i++) {
            String arg = args[i];
            
            if (arg.equals("-n")) {
                noNewline = true;
            } else if (arg.equals(">") && i + 1 < args.length) {
                outputFile = args[++i];
            } else if (arg.equals(">>") && i + 1 < args.length) {
                outputFile = args[++i];
                // Mode append (géré plus tard)
            } else {
                if (text.length() > 0) {
                    text.append(" ");
                }
                text.append(arg);
            }
        }
        
        String content = text.toString();
        
        if (outputFile != null) {
            // Écrire dans un fichier
            try {
                String fileName = PathResolver.getBaseName(outputFile);
                String parentPath = PathResolver.getParentPath(outputFile);
                
                VirtualDirectory parentDir;
                if (parentPath.equals("/")) {
                    parentDir = fileSystem.getRoot();
                } else {
                    VirtualFile parent = fileSystem.resolvePath(parentPath);
                    if (parent instanceof VirtualDirectory) {
                        parentDir = (VirtualDirectory) parent;
                    } else {
                        return "Erreur: " + parentPath + " n'est pas un dossier";
                    }
                }
                
                VirtualFile file = parentDir.getFile(fileName);
                
                if (file == null) {
                    file = new VirtualFile(fileName);
                    parentDir.addFile(file);
                }
                
                file.setContent(content);
                return "";
            } catch (FileSystemException e) {
                return "Erreur: " + e.getMessage();
            }
        } else {
            // Afficher à l'écran
            if (noNewline) {
                return content;
            } else {
```

```java
package fr.terminal.emulator.commands;

import fr.terminal.emulator.filesystem.*;
import java.util.List;

/**
 * Commande 'tree' : Affiche l'arborescence des fichiers.
 */
public class TreeCommand implements Command {
    
    private final VirtualFileSystem fileSystem;
    
    /**
     * Crée la commande tree.
     * @param fileSystem Le système de fichiers
     */
    public TreeCommand(VirtualFileSystem fileSystem) {
        this.fileSystem = fileSystem;
    }
    
    @Override
    public String getName() {
        return "tree";
    }
    
    @Override
    public String getDescription() {
        return "Affiche l'arborescence des fichiers et dossiers";
    }
    
    @Override
    public String getUsage() {
        return "tree [depth]";
    }
    
    @Override
    public String execute(String[] args) {
        int maxDepth = Integer.MAX_VALUE;
        
        if (args.length > 0) {
            try {
                maxDepth = Integer.parseInt(args[0]);
            } catch (NumberFormatException e) {
                return "Erreur: " + args[0] + " n'est pas un nombre valide";
            }
        }
        
        VirtualDirectory root = fileSystem.getCurrentDirectory();
        StringBuilder sb = new StringBuilder();
        
        buildTree(root, "", true, 0, maxDepth, sb);
        
        return sb.toString();
    }
    
    /**
     * Construit récursivement l'arborescence.
     * @param directory Le dossier à afficher
     * @param prefix Le préfixe pour l'indentation
     * @param isLast Indique si c'est le dernier élément
     * @param depth La profondeur actuelle
     * @param maxDepth La profondeur maximale
     * @param sb Le StringBuilder pour le résultat
     */
    private void buildTree(VirtualDirectory directory, String prefix, boolean isLast, 
                          int depth, int maxDepth, StringBuilder sb) {
        if (depth > maxDepth) {
            return;
        }
        
        // Afficher le dossier courant
        sb.append(prefix);
        sb.append(isLast ? "└── " : "├── ");
        sb.append(directory.getName());
        sb.append("\n");
        
        // Si on a atteint la profondeur maximale, on s'arrête
        if (depth >= maxDepth) {
            return;
        }
        
        // Préparer le nouveau préfixe
        String newPrefix = prefix + (isLast ? "    " : "│   ");
        
        // Lister tous les éléments (dossiers d'abord, puis fichiers)
        List<String> allItems = directory.listContents();
        
        // Séparer dossiers et fichiers
        java.util.List<String> dirs = new java.util.ArrayList<>();
        java.util.List<String> files = new java.util.ArrayList<>();
        
        for (String item : allItems) {
            if (directory.getDirectory(item) != null) {
                dirs.add(item);
            } else {
                files.add(item);
            }
        }
        
        // Afficher les dossiers
        for (int i = 0; i < dirs.size(); i++) {
            String dirName = dirs.get(i);
            VirtualDirectory subDir = directory.getDirectory(dirName);
            boolean last = (i == dirs.size() - 1 && files.isEmpty()) || 
                          (i == dirs.size() - 1 && !files.isEmpty());
            buildTree(subDir, newPrefix, last, depth + 1, maxDepth, sb);
        }
        
        // Afficher les fichiers
        for (int i = 0; i < files.size(); i++) {
            String fileName = files.get(i);
            sb.append(newPrefix);
            sb.append((dirs.isEmpty() && i == files.size() - 1) || 
                     (!dirs.isEmpty() && i == files.size() - 1) ? "└── " : "├── ");
            sb.append(fileName);
            sb.append("\n");
        }
    }
}
```

```java
package fr.terminal.emulator.utils;

import java.util.ArrayList;
import java.util.List;

/**
 * Gère l'historique des commandes saisies par l'utilisateur.
 */
public class CommandHistory {
    
    private final List<String> history;
    private int currentIndex;
    
    /**
     * Crée un nouvel historique de commandes.
     */
    public CommandHistory() {
        this.history = new ArrayList<>();
        this.currentIndex = -1;
    }
    
    /**
     * Ajoute une commande à l'historique.
     * @param command La commande à ajouter
     */
    public void add(String command) {
        if (command != null && !command.trim().isEmpty()) {
            // Ne pas ajouter de doublons consécutifs
            if (!history.isEmpty() && history.get(history.size() - 1).equals(command)) {
                return;
            }
            history.add(command);
            currentIndex = history.size();
        }
    }
    
    /**
     * Donne la commande précédente dans l'historique.
     * @return La commande précédente, ou null si on est au début
     */
    public String getPrevious() {
        if (history.isEmpty()) {
            return null;
        }
        
        if (currentIndex > 0) {
            currentIndex--;
            return history.get(currentIndex);
        } else {
            return null;
        }
    }
    
    /**
     * Donne la commande suivante dans l'historique.
     * @return La commande suivante, ou null si on est à la fin
     */
    public String getNext() {
        if (history.isEmpty()) {
            return null;
        }
        
        if (currentIndex < history.size() - 1) {
            currentIndex++;
            return history.get(currentIndex);
        } else {
            currentIndex = history.size();
            return null;
        }
    }
    
    /**
     * Réinitialise l'index pour une nouvelle saisie.
     */
    public void resetIndex() {
        currentIndex = history.size();
    }
    
    /**
     * Donne toute l'historique.
     * @return Liste de toutes les commandes
     */
    public List<String> getHistory() {
        return new ArrayList<>(history);
    }
    
    /**
     * Efface l'historique.
     */
    public void clear() {
        history.clear();
        currentIndex = -1;
    }
    
    /**
     * Donne le nombre de commandes dans l'historique.
     * @return Le nombre de commandes
     */
    public int size() {
        return history.size();
    }
}
```

## 📚 Documentation Technique

### Concepts Clés

#### 1. **Système de Fichiers Virtuel**

Le système de fichiers est entièrement implémenté en mémoire avec :
- `VirtualFileSystem` : Gère la racine et le répertoire courant
- `VirtualDirectory` : Représente un dossier avec ses sous-éléments
- `VirtualFile` : Représente un fichier avec son contenu
- `FileSystemException` : Pour la gestion des erreurs

**Caractéristiques** :
- Hiérarchie arborescente classique
- Chemins absolus (`/home/user`) et relatifs (`../parent`)
- Gestion des noms de fichiers/dossiers uniques dans un même répertoire

#### 2. **Interpréteur de Commandes**

L'interpréteur (`CommandInterpreter`) :
- Parse la ligne de commande en tokens
- Résout les chemins de fichiers
- Trouve et exécute la commande appropriée
- Gère les erreurs et retourne les résultats

**Architecture des commandes** :
- Chaque commande implémente l'interface `Command`
- Méthodes requises : `getName()`, `getDescription()`, `getUsage()`, `execute(String[])`
- Les commandes reçoivent le système de fichiers via injection de dépendance

#### 3. **Interface Graphique Swing**

L'interface utilise :
- `TerminalFrame` : Fenêtre principale
- `TerminalPanel` : Panel principal avec entrée/sortie
- `TerminalOutput` : Zone de texte pour l'affichage (avec couleurs)
- `TerminalInput` : Champ de saisie pour les commandes

**Fonctionnalités UI** :
- Thème sombre par défaut
- Police monospace
- Historique des commandes (↑/↓)
- Défilement automatique
- Prompt personnalisable

#### 4. **Gestion des Chemins**

`PathResolver` fournit des utilitaires pour :
- Normalisation des chemins (`//` → `/`, `\` → `/`)
- Extraction du nom de base (`/home/user/file.txt` → `file.txt`)
- Extraction du parent (`/home/user/file.txt` → `/home/user`)
- Détection des chemins absolus vs relatifs

### 1. Visualisation Desktop (Priorité 1)

Pour ajouter une visualisation Desktop comme demandé :

```java
// Nouvelle classe DesktopView.java
package fr.terminal.emulator;

import javax.swing.*;
import java.awt.*;
import fr.terminal.emulator.filesystem.*;

/**
 * Vue Desktop pour visualiser les fichiers sous forme d'icônes.
 */
public class DesktopView extends JPanel {
    
    private final VirtualFileSystem fileSystem;
    private final int iconSize = 64;
    private final int margin = 20;
    
    public DesktopView(VirtualFileSystem fileSystem) {
        this.fileSystem = fileSystem;
        setLayout(new FlowLayout(FlowLayout.LEFT, margin, margin));
        setBackground(new Color(40, 40, 40));
        refresh();
    }
    
    /**
     * Rafraîchit l'affichage avec le contenu du répertoire courant.
     */
    public void refresh() {
        removeAll();
        
        VirtualDirectory current = fileSystem.getCurrentDirectory();
        
        // Ajouter l'icône "Parent"
        if (current.getParent() != null) {
            add(createIconPanel("../", "dossier_parent.png", () -> {
                fileSystem.setCurrentDirectory(current.getParent());
                refresh();
            }));
        }
        
        // Ajouter les icônes des sous-dossiers
        for (VirtualDirectory dir : current.getDirectories()) {
            add(createIconPanel(dir.getName() + "/", "dossier.png", () -> {
                fileSystem.setCurrentDirectory(dir);
                refresh();
            }));
        }
        
        // Ajouter les icônes des fichiers
        for (VirtualFile file : current.getFiles()) {
            add(createIconPanel(file.getName(), "fichier.png", () -> {
                // Ouvrir le fichier (à implémenter)
                JOptionPane.showMessageDialog(this, 
                    file.getContent(), file.getName(), 
                    JOptionPane.INFORMATION_MESSAGE);
            }));
        }
        
        revalidate();
        repaint();
    }
    
    private JPanel createIconPanel(String label, String iconName, Runnable action) {
        JPanel panel = new JPanel();
        panel.setLayout(new BorderLayout());
        panel.setPreferredSize(new Dimension(iconSize + 20, iconSize + 40));
        panel.setBackground(new Color(40, 40, 40));
        
        // Icône (simplifiée - en vrai on utiliserait ImageIO)
        JLabel iconLabel = new JLabel("📁", SwingConstants.CENTER);
        iconLabel.setFont(new Font("Segoe UI Emoji", Font.PLAIN, 40));
        iconLabel.setPreferredSize(new Dimension(iconSize, iconSize));
        
        // Label du nom
        JLabel nameLabel = new JLabel(label, SwingConstants.CENTER);
        nameLabel.setForeground(Color.WHITE);
        nameLabel.setFont(new Font("SansSerif", Font.PLAIN, 12));
        
        panel.add(iconLabel, BorderLayout.CENTER);
        panel.add(nameLabel, BorderLayout.SOUTH);
        
        // Gestion du clic
        panel.setCursor(new Cursor(Cursor.HAND_CURSOR));
        panel.addMouseListener(new java.awt.event.MouseAdapter() {
            @Override
            public void mouseClicked(java.awt.event.MouseEvent e) {
                if (e.getClickCount() == 2) {
                    action.run();
                }
            }
        });
        
        return panel;
    }
}
```

```java
// Modifier TerminalFrame.java
public class TerminalFrame extends JFrame {
    private TerminalPanel terminalPanel;
    private DesktopView desktopView;
    private JTabbedPane tabbedPane;
    
    public TerminalFrame() {
        // ... configuration de base
        
        tabbedPane = new JTabbedPane();
        
        // Onglet Terminal
        terminalPanel = new TerminalPanel();
        tabbedPane.addTab("Terminal", terminalPanel);
        
        // Onglet Desktop
        VirtualFileSystem fs = terminalPanel.getCommandInterpreter().getFileSystem();
        desktopView = new DesktopView(fs);
        tabbedPane.addTab("Desktop", new JScrollPane(desktopView));
        
        add(tabbedPane, BorderLayout.CENTER);
        
        // Synchroniser les changements de répertoire
        terminalPanel.getCommandInterpreter().getFileSystem()
            .addChangeListener(() -> desktopView.refresh());
    }
}
```

#### **TerminalPanel.java**

```java
package gabywald.terminal;

import gabywald.terminal.shell.Shell;
import gabywald.terminal.shell.CommandResult;
import gabywald.terminal.model.TerminalState;
import gabywald.terminal.model.CommandHistory;
import gabywald.terminal.filesystem.FileNode;

import javax.swing.*;
import java.awt.*;
import java.awt.event.*;
import java.util.List;

public class TerminalPanel extends JPanel {
    private static final long serialVersionUID = 1L;
    private static final String PROMPT = "terminal> ";
    private static final int MARGIN = 10;
    private static final int MAX_LINES = 1000;

    private final TerminalTextArea textArea;
    private final Shell shell;
    private final CommandHistory commandHistory;
    private final TerminalState terminalState;

    private String currentInput = "";
    private int historyIndex = -1;
    private int cursorPosition = 0;

    public TerminalPanel() {
        setLayout(new BorderLayout());
        setBackground(Color.BLACK);
        setForeground(Color.WHITE);

        this.shell = new Shell();
        this.commandHistory = new CommandHistory(100);
        this.terminalState = new TerminalState();

        this.textArea = new TerminalTextArea();
        textArea.setEditable(false);
        textArea.setBackground(Color.BLACK);
        textArea.setForeground(Color.WHITE);
        textArea.setFont(new Font("Monospaced", Font.PLAIN, 14));
        textArea.setMargin(new Insets(MARGIN, MARGIN, MARGIN, MARGIN));
        textArea.setCaretColor(Color.WHITE);

        JScrollPane scrollPane = new JScrollPane(textArea);
        scrollPane.setBorder(BorderFactory.createEmptyBorder());
        scrollPane.getVerticalScrollBar().setUnitIncrement(16);
        add(scrollPane, BorderLayout.CENTER);

        addKeyListener();
        displayWelcomeMessage();
        displayPrompt();
    }

    private void addKeyListener() {
        addKeyListener(new KeyAdapter() {
            @Override
            public void keyPressed(KeyEvent e) {
                handleKeyPressed(e);
            }
        });
        setFocusable(true);
        requestFocusInWindow();
    }

    private void handleKeyPressed(KeyEvent e) {
        int keyCode = e.getKeyCode();
        char keyChar = e.getKeyChar();

        switch (keyCode) {
            case KeyEvent.VK_ENTER:
                handleEnter();
                break;
            case KeyEvent.VK_BACK_SPACE:
                handleBackspace();
                break;
            case KeyEvent.VK_DELETE:
                handleDelete();
                break;
            case KeyEvent.VK_LEFT:
                handleLeftArrow();
                break;
            case KeyEvent.VK_RIGHT:
                handleRightArrow();
                break;
            case KeyEvent.VK_UP:
                handleUpArrow();
                break;
            case KeyEvent.VK_DOWN:
                handleDownArrow();
                break;
            case KeyEvent.VK_HOME:
                handleHome();
                break;
            case KeyEvent.VK_END:
                handleEnd();
                break;
            case KeyEvent.VK_TAB:
                handleTab();
                break;
            case KeyEvent.VK_C:
                if (e.isControlDown()) handleCopy();
                break;
            case KeyEvent.VK_V:
                if (e.isControlDown()) handlePaste();
                break;
            default:
                if (Character.isPrintable(keyChar) && !e.isControlDown()) {
                    handlePrintableKey(keyChar);
                }
                break;
        }
        e.consume();
    }

    private void handleEnter() {
        if (currentInput.isEmpty()) {
            textArea.append("\n");
            displayPrompt();
            return;
        }

        commandHistory.add(currentInput);
        historyIndex = -1;

        textArea.append(currentInput + "\n");
        CommandResult result = shell.execute(currentInput);

        if (result != null) {
            if (result.getOutput() != null && !result.getOutput().isEmpty()) {
                textArea.append(result.getOutput() + "\n");
            }
            if (result.getError() != null && !result.getError().isEmpty()) {
                textArea.append("Error: " + result.getError() + "\n");
            }
        }

        if (currentInput.equals("clear")) {
            clear();
            return;
        }

        currentInput = "";
        cursorPosition = 0;
        displayPrompt();
    }

    private void handleBackspace() {
        if (cursorPosition > 0) {
            currentInput = currentInput.substring(0, cursorPosition - 1)
                    + currentInput.substring(cursorPosition);
            cursorPosition--;
            updateInputDisplay();
        }
    }

    private void handleDelete() {
        if (cursorPosition < currentInput.length()) {
            currentInput = currentInput.substring(0, cursorPosition)
                    + currentInput.substring(cursorPosition + 1);
            updateInputDisplay();
        }
    }

    private void handleLeftArrow() {
        if (cursorPosition > 0) {
            cursorPosition--;
            updateInputDisplay();
        }
    }

    private void handleRightArrow() {
        if (cursorPosition < currentInput.length()) {
            cursorPosition++;
            updateInputDisplay();
        }
    }

    private void handleUpArrow() {
        List<String> history = commandHistory.getHistory();
        if (history.isEmpty()) return;

        if (historyIndex == -1) {
            historyIndex = history.size() - 1;
        } else if (historyIndex > 0) {
            historyIndex--;
        }

        if (historyIndex >= 0 && historyIndex < history.size()) {
            currentInput = history.get(historyIndex);
            cursorPosition = currentInput.length();
            updateInputDisplay();
        }
    }

    private void handleDownArrow() {
        List<String> history = commandHistory.getHistory();
        if (history.isEmpty()) return;

        if (historyIndex != -1 && historyIndex < history.size() - 1) {
            historyIndex++;
            currentInput = history.get(historyIndex);
            cursorPosition = currentInput.length();
            updateInputDisplay();
        } else {
            historyIndex = -1;
            currentInput = "";
            cursorPosition = 0;
            updateInputDisplay();
        }
    }

    private void handleHome() {
        cursorPosition = 0;
        updateInputDisplay();
    }

    private void handleEnd() {
        cursorPosition = currentInput.length();
        updateInputDisplay();
    }

    private void handleTab() {
        String[] parts = currentInput.split("\\s+");
        if (parts.length == 0) return;

        String lastPart = parts[parts.length - 1];
        String prefix = currentInput.substring(0, currentInput.length() - lastPart.length());

        List<String> suggestions = shell.getSuggestions(lastPart);

        if (!suggestions.isEmpty()) {
            String suggestion = suggestions.get(0);
            currentInput = prefix + suggestion + " ";
            cursorPosition = currentInput.length();
            updateInputDisplay();
        }
    }

    private void handleCopy() {
        String selection = textArea.getSelectedText();
        if (selection != null && !selection.isEmpty()) {
            StringSelection stringSelection = new StringSelection(selection);
            Toolkit.getDefaultToolkit().getSystemClipboard()
                    .setContents(stringSelection, null);
        }
    }

    private void handlePaste() {
        try {
            String clipboardContent = (String) Toolkit.getDefaultToolkit()
                    .getSystemClipboard().getData(DataFlavor.stringFlavor);
            if (clipboardContent != null) {
                currentInput = currentInput.substring(0, cursorPosition)
                        + clipboardContent
                        + currentInput.substring(cursorPosition);
                cursorPosition += clipboardContent.length();
                updateInputDisplay();
            }
        } catch (Exception e) {
            // Ignorer
        }
    }

    private void handlePrintableKey(char keyChar) {
        currentInput = currentInput.substring(0, cursorPosition)
                + keyChar
                + currentInput.substring(cursorPosition);
        cursorPosition++;
        updateInputDisplay();
    }

    private void updateInputDisplay() {
        int currentLength = textArea.getDocument().getLength();
        int promptLength = PROMPT.length();
        int lineStart = currentLength - (currentInput.length() + promptLength);

        if (lineStart >= 0) {
            try {
                textArea.getDocument().remove(lineStart, currentLength - lineStart);
            } catch (Exception e) {
                e.printStackTrace();
            }
        }

        textArea.append(PROMPT + currentInput);
        textArea.setCaretPosition(textArea.getDocument().getLength());
    }

    private void displayPrompt() {
        textArea.append(PROMPT);
        textArea.setCaretPosition(textArea.getDocument().getLength());
    }

    private void displayWelcomeMessage() {
        String welcomeMessage =
                "==============================================\n"
                        + "  Terminal Emulator - Unix/Linux Style      \n"
                        + "  Version: 1.0.0                            \n"
                        + "  Auteur: Gabriel Chandesris                \n"
                        + "  Date: 30 Juin 2026                        \n"
                        + "==============================================\n\n"
                        + "Tapez 'help' pour la liste des commandes.\n"
                        + "Utilisez UP/DOWN pour naviguer dans l'historique.\n"
                        + "Utilisez Tab pour l'auto-completion.\n\n";
        textArea.append(welcomeMessage);
    }

    public void appendText(String text) {
        textArea.append(text);
    }

    public void clear() {
        textArea.setText("");
        displayWelcomeMessage();
        displayPrompt();
    }

    public void copy() {
        handleCopy();
    }

    public void paste() {
        handlePaste();
    }

    public Shell getShell() {
        return shell;
    }

    public CommandHistory getCommandHistory() {
        return commandHistory;
    }

    public TerminalState getTerminalState() {
        return terminalState;
    }
}
```

```java
package gabywald.terminal;

import javax.swing.*;
import javax.swing.text.*;
import java.awt.*;

public class TerminalTextArea extends JTextArea {
    private static final long serialVersionUID = 1L;
    private static final int MAX_LINES = 1000;

    public TerminalTextArea() {
        super();
        setEditable(false);
        setFont(new Font("Monospaced", Font.PLAIN, 14));
        setTabSize(4);
        setLineWrap(true);
        setWrapStyleWord(true);

        getDocument().addDocumentListener(new DocumentListener() {
            @Override
            public void insertUpdate(DocumentEvent e) {
                limitLines();
            }
            @Override
            public void removeUpdate(DocumentEvent e) {}
            @Override
            public void changedUpdate(DocumentEvent e) {}
        });
    }

    private void limitLines() {
        Document doc = getDocument();
        Element root = doc.getDefaultRootElement();

        while (root.getElementCount() > MAX_LINES) {
            try {
                int end = root.getElement(0).getEndOffset();
                doc.remove(0, end + 1);
            } catch (BadLocationException e) {
                break;
            }
        }
    }

    @Override
    protected void paintComponent(Graphics g) {
        g.setColor(getBackground());
        g.fillRect(0, 0, getWidth(), getHeight());
        super.paintComponent(g);
    }

    @Override
    public void setFont(Font font) {
        super.setFont(font);
        if (!font.getFamily().equals("Monospaced")) {
            super.setFont(new Font("Monospaced", font.getStyle(), font.getSize()));
        }
    }
}
```

```java
package gabywald.terminal.shell;

public class CommandResult {
    private final String output;
    private final String error;
    private final int exitCode;

    public CommandResult(String output, int exitCode) {
        this(output, null, exitCode);
    }

    public CommandResult(String output, String error, int exitCode) {
        this.output = output != null ? output : "";
        this.error = error != null ? error : "";
        this.exitCode = exitCode;
    }

    public CommandResult(String error) {
        this(null, error, 1);
    }

    public CommandResult() {
        this("", 0);
    }

    public String getOutput() { return output; }
    public String getError() { return error; }
    public int getExitCode() { return exitCode; }
    public boolean isSuccess() { return exitCode == 0; }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        if (!output.isEmpty()) sb.append(output);
        if (!error.isEmpty()) {
            if (sb.length() > 0) sb.append("\n");
            sb.append("Error: ").append(error);
        }
        return sb.toString();
    }
}
```
