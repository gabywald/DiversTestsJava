package com.terminal.emulator;

import com.terminal.emulator.filesystem.Directory;
import java.util.ArrayList;
import java.util.List;

/**
 * Classe représentant l'état global du terminal.
 * Contient le répertoire courant, l'historique des commandes, etc.
 * 
 * @author Gabriel Chandesris
 * @version 1.0.0
 */
public class TerminalState {
    
    private Directory currentDirectory;
    private Directory rootDirectory;
    private List<String> commandHistory;
    private int historyIndex;
    private String prompt;
    
    /**
     * Constructeur par défaut.
     * Initialise le système de fichiers avec un répertoire racine.
     */
    public TerminalState() {
        this.rootDirectory = new Directory("/", null);
        this.currentDirectory = rootDirectory;
        this.commandHistory = new ArrayList<>();
        this.historyIndex = -1;
        this.prompt = "user@terminal:~$";
    }
    
    /**
     * Constructeur avec un répertoire racine personnalisé.
     * 
     * @param root Répertoire racine du système de fichiers
     */
    public TerminalState(Directory root) {
        this.rootDirectory = root;
        this.currentDirectory = root;
        this.commandHistory = new ArrayList<>();
        this.historyIndex = -1;
        this.prompt = "user@terminal:~$";
    }
    
    // Getters et Setters
    
    /**
     * @return Le répertoire courant
     */
    public Directory getCurrentDirectory() {
        return currentDirectory;
    }
    
    /**
     * Définit le répertoire courant.
     * 
     * @param currentDirectory Nouveau répertoire courant
     */
    public void setCurrentDirectory(Directory currentDirectory) {
        this.currentDirectory = currentDirectory;
        updatePrompt();
    }
    
    /**
     * @return Le répertoire racine
     */
    public Directory getRootDirectory() {
        return rootDirectory;
    }
    
    /**
     * @return L'historique des commandes
     */
    public List<String> getCommandHistory() {
        return new ArrayList<>(commandHistory);
    }
    
    /**
     * Ajoute une commande à l'historique.
     * 
     * @param command Commande à ajouter
     */
    public void addToHistory(String command) {
        if (command != null && !command.trim().isEmpty()) {
            commandHistory.add(command);
            historyIndex = commandHistory.size();
        }
    }
    
    /**
     * @return L'index courant dans l'historique
     */
    public int getHistoryIndex() {
        return historyIndex;
    }
    
    /**
     * Définit l'index de l'historique.
     * 
     * @param historyIndex Nouveau index
     */
    public void setHistoryIndex(int historyIndex) {
        this.historyIndex = historyIndex;
    }
    
    /**
     * @return Le prompt actuel
     */
    public String getPrompt() {
        return prompt;
    }
    
    /**
     * Met à jour le prompt en fonction du répertoire courant.
     */
    private void updatePrompt() {
        String path = currentDirectory.getPath();
        if ("/".equals(path)) {
            prompt = "user@terminal:~$";
        } else {
            prompt = "user@terminal:" + path + "$";
        }
    }
    
    /**
     * Récupère la commande précédente dans l'historique.
     * 
     * @return Commande précédente ou null
     */
    public String getPreviousCommand() {
        if (commandHistory.isEmpty()) {
            return null;
        }
        
        if (historyIndex > 0) {
            historyIndex--;
            return commandHistory.get(historyIndex);
        } else if (historyIndex == 0) {
            return commandHistory.get(0);
        }
        
        return null;
    }
    
    /**
     * Récupère la commande suivante dans l'historique.
     * 
     * @return Commande suivante ou null
     */
    public String getNextCommand() {
        if (commandHistory.isEmpty()) {
            return null;
        }
        
        if (historyIndex < commandHistory.size() - 1) {
            historyIndex++;
            return commandHistory.get(historyIndex);
        } else if (historyIndex == commandHistory.size() - 1) {
            historyIndex = commandHistory.size();
            return "";
        }
        
        return null;
    }
    
    /**
     * Réinitialise l'index de l'historique.
     */
    public void resetHistoryIndex() {
        historyIndex = commandHistory.size();
    }
}

package com.terminal.emulator.filesystem;

import java.time.LocalDateTime;

/**
 * Classe abstraite représentant un nœud dans le système de fichiers.
 * Peut être un fichier ou un répertoire.
 * 
 * @author Gabriel Chandesris
 * @version 1.0.0
 */
public abstract class FileNode {
    
    protected String name;
    protected Directory parent;
    protected LocalDateTime createdAt;
    protected LocalDateTime modifiedAt;
    
    /**
     * Constructeur principal.
     * 
     * @param name Nom du nœud
     * @param parent Répertoire parent
     */
    public FileNode(String name, Directory parent) {
        this.name = name;
        this.parent = parent;
        this.createdAt = LocalDateTime.now();
        this.modifiedAt = LocalDateTime.now();
    }
    
    /**
     * @return Le nom du nœud
     */
    public String getName() {
        return name;
    }
    
    /**
     * Définit le nom du nœud.
     * 
     * @param name Nouveau nom
     */
    public void setName(String name) {
        this.name = name;
        this.modifiedAt = LocalDateTime.now();
    }
    
    /**
     * @return Le répertoire parent
     */
    public Directory getParent() {
        return parent;
    }
    
    /**
     * Définit le répertoire parent.
     * 
     * @param parent Nouveau parent
     */
    public void setParent(Directory parent) {
        this.parent = parent;
        this.modifiedAt = LocalDateTime.now();
    }
    
    /**
     * @return La date de création
     */
    public LocalDateTime getCreatedAt() {
        return createdAt;
    }
    
    /**
     * @return La date de dernière modification
     */
    public LocalDateTime getModifiedAt() {
        return modifiedAt;
    }
    
    /**
     * @return Le chemin absolu du nœud
     */
    public abstract String getPath();
    
    /**
     * @return True si c'est un répertoire
     */
    public abstract boolean isDirectory();
    
    /**
     * @return True si c'est un fichier
     */
    public abstract boolean isFile();
    
    /**
     * @return La taille du nœud (0 pour un répertoire)
     */
    public abstract long getSize();
    
    /**
     * Supprime ce nœud.
     * 
     * @return True si la suppression a réussi
     */
    public abstract boolean delete();
    
    /**
     * @return Une représentation textuelle du nœud
     */
    @Override
    public String toString() {
        return name;
    }
    
    /**
     * Compare deux nœuds par nom.
     * 
     * @param other Autre nœud
     * @return Résultat de la comparaison
     */
    @Override
    public int compareTo(FileNode other) {
        return this.name.compareTo(other.name);
    }
    
    /**
     * Vérifie l'égalité avec un autre objet.
     * 
     * @param obj Objet à comparer
     * @return True si égal
     */
    @Override
    public boolean equals(Object obj) {
        if (this == obj) return true;
        if (obj == null || getClass() != obj.getClass()) return false;
        FileNode fileNode = (FileNode) obj;
        return name.equals(fileNode.name) && 
               (parent == null ? fileNode.parent == null : parent.equals(fileNode.parent));
    }
    
    /**
     * @return Code de hachage
     */
    @Override
    public int hashCode() {
        int result = name.hashCode();
        result = 31 * result + (parent != null ? parent.hashCode() : 0);
        return result;
    }
}

package com.terminal.emulator.filesystem;

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.stream.Collectors;

/**
 * Classe représentant un répertoire dans le système de fichiers.
 * 
 * @author Gabriel Chandesris
 * @version 1.0.0
 */
public class Directory extends FileNode implements Comparable<FileNode> {
    
    private List<FileNode> children;
    
    /**
     * Constructeur principal.
     * 
     * @param name Nom du répertoire
     * @param parent Répertoire parent
     */
    public Directory(String name, Directory parent) {
        super(name, parent);
        this.children = new ArrayList<>();
    }
    
    /**
     * @return La liste des enfants (fichiers et sous-répertoires)
     */
    public List<FileNode> getChildren() {
        return Collections.unmodifiableList(children);
    }
    
    /**
     * Ajoute un enfant à ce répertoire.
     * 
     * @param child Nœud à ajouter
     * @return True si l'ajout a réussi
     */
    public boolean addChild(FileNode child) {
        if (child == null) return false;
        
        // Vérifier si un enfant avec le même nom existe déjà
        for (FileNode existing : children) {
            if (existing.getName().equals(child.getName())) {
                return false;
            }
        }
        
        child.setParent(this);
        children.add(child);
        this.modifiedAt = java.time.LocalDateTime.now();
        return true;
    }
    
    /**
     * Supprime un enfant de ce répertoire.
     * 
     * @param child Nœud à supprimer
     * @return True si la suppression a réussi
     */
    public boolean removeChild(FileNode child) {
        if (child == null) return false;
        
        boolean removed = children.remove(child);
        if (removed) {
            child.setParent(null);
            this.modifiedAt = java.time.LocalDateTime.now();
        }
        return removed;
    }
    
    /**
     * Supprime un enfant par son nom.
     * 
     * @param name Nom du nœud à supprimer
     * @return True si la suppression a réussi
     */
    public boolean removeChildByName(String name) {
        FileNode toRemove = getChild(name);
        if (toRemove != null) {
            return removeChild(toRemove);
        }
        return false;
    }
    
    /**
     * Récupère un enfant par son nom.
     * 
     * @param name Nom du nœud
     * @return Le nœud trouvé ou null
     */
    public FileNode getChild(String name) {
        for (FileNode child : children) {
            if (child.getName().equals(name)) {
                return child;
            }
        }
        return null;
    }
    
    /**
     * Vérifie si un enfant avec le nom donné existe.
     * 
     * @param name Nom à vérifier
     * @return True si existe
     */
    public boolean hasChild(String name) {
        return getChild(name) != null;
    }
    
    /**
     * Récupère tous les sous-répertoires.
     * 
     * @return Liste des sous-répertoires
     */
    public List<Directory> getSubdirectories() {
        return children.stream()
                .filter(FileNode::isDirectory)
                .map(node -> (Directory) node)
                .collect(Collectors.toList());
    }
    
    /**
     * Récupère tous les fichiers.
     * 
     * @return Liste des fichiers
     */
    public List<TerminalFile> getFiles() {
        return children.stream()
                .filter(FileNode::isFile)
                .map(node -> (TerminalFile) node)
                .collect(Collectors.toList());
    }
    
    /**
     * @return Le chemin absolu du répertoire
     */
    @Override
    public String getPath() {
        if (parent == null) {
            return "/";
        }
        
        String parentPath = parent.getPath();
        if ("/".equals(parentPath)) {
            return "/" + name;
        }
        return parentPath + "/" + name;
    }
    
    /**
     * @return True (c'est un répertoire)
     */
    @Override
    public boolean isDirectory() {
        return true;
    }
    
    /**
     * @return False (ce n'est pas un fichier)
     */
    @Override
    public boolean isFile() {
        return false;
    }
    
    /**
     * @return 0 (un répertoire n'a pas de taille)
     */
    @Override
    public long getSize() {
        return 0;
    }
    
    /**
     * Supprime ce répertoire et tout son contenu.
     * 
     * @return True si la suppression a réussi
     */
    @Override
    public boolean delete() {
        if (parent != null) {
            return parent.removeChild(this);
        }
        return false;
    }
    
    /**
     * Supprime tout le contenu du répertoire.
     */
    public void clear() {
        children.clear();
        this.modifiedAt = java.time.LocalDateTime.now();
    }
    
    /**
     * @return Le nombre d'enfants
     */
    public int getChildCount() {
        return children.size();
    }
    
    /**
     * @return True si le répertoire est vide
     */
    public boolean isEmpty() {
        return children.isEmpty();
    }
    
    /**
     * Crée un sous-répertoire.
     * 
     * @param name Nom du sous-répertoire
     * @return Le nouveau répertoire ou null si déjà existe
     */
    public Directory createDirectory(String name) {
        if (hasChild(name)) {
            return null;
        }
        
        Directory dir = new Directory(name, this);
        addChild(dir);
        return dir;
    }
    
    /**
     * Crée un fichier.
     * 
     * @param name Nom du fichier
     * @return Le nouveau fichier ou null si déjà existe
     */
    public TerminalFile createFile(String name) {
        if (hasChild(name)) {
            return null;
        }
        
        TerminalFile file = new TerminalFile(name, this);
        addChild(file);
        return file;
    }
    
    /**
     * @return Une représentation textuelle détaillée
     */
    @Override
    public String toString() {
        return "[DIR] " + name + " (" + children.size() + " items)";
    }
}

package com.terminal.emulator.filesystem;

/**
 * Classe représentant un fichier texte dans le système de fichiers.
 * 
 * @author Gabriel Chandesris
 * @version 1.0.0
 */
public class TerminalFile extends FileNode implements Comparable<FileNode> {
    
    private String content;
    
    /**
     * Constructeur principal.
     * 
     * @param name Nom du fichier
     * @param parent Répertoire parent
     */
    public TerminalFile(String name, Directory parent) {
        super(name, parent);
        this.content = "";
    }
    
    /**
     * Constructeur avec contenu initial.
     * 
     * @param name Nom du fichier
     * @param parent Répertoire parent
     * @param content Contenu initial
     */
    public TerminalFile(String name, Directory parent, String content) {
        super(name, parent);
        this.content = content != null ? content : "";
    }
    
    /**
     * @return Le contenu du fichier
     */
    public String getContent() {
        return content;
    }
    
    /**
     * Définit le contenu du fichier.
     * 
     * @param content Nouveau contenu
     */
    public void setContent(String content) {
        this.content = content != null ? content : "";
        this.modifiedAt = java.time.LocalDateTime.now();
    }
    
    /**
     * Ajoute du contenu au fichier.
     * 
     * @param text Texte à ajouter
     */
    public void appendContent(String text) {
        if (text != null) {
            this.content += text;
            this.modifiedAt = java.time.LocalDateTime.now();
        }
    }
    
    /**
     * @return Le chemin absolu du fichier
     */
    @Override
    public String getPath() {
        if (parent == null) {
            return "/" + name;
        }
        
        String parentPath = parent.getPath();
        if ("/".equals(parentPath)) {
            return "/" + name;
        }
        return parentPath + "/" + name;
    }
    
    /**
     * @return False (ce n'est pas un répertoire)
     */
    @Override
    public boolean isDirectory() {
        return false;
    }
    
    /**
     * @return True (c'est un fichier)
     */
    @Override
    public boolean isFile() {
        return true;
    }
    
    /**
     * @return La taille du fichier en octets
     */
    @Override
    public long getSize() {
        return content.getBytes().length;
    }
    
    /**
     * Supprime ce fichier.
     * 
     * @return True si la suppression a réussi
     */
    @Override
    public boolean delete() {
        if (parent != null) {
            return parent.removeChild(this);
        }
        return false;
    }
    
    /**
     * Vide le contenu du fichier.
     */
    public void clear() {
        this.content = "";
        this.modifiedAt = java.time.LocalDateTime.now();
    }
    
    /**
     * @return True si le fichier est vide
     */
    public boolean isEmpty() {
        return content.isEmpty();
    }
    
    /**
     * @return Une représentation textuelle détaillée
     */
    @Override
    public String toString() {
        return "[FILE] " + name + " (" + getSize() + " bytes)";
    }
}

// DESCRIPTION : Parseur de commandes pour extraire le nom et les arguments

package com.terminal.emulator.commands;

import java.util.ArrayList;
import java.util.List;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * Parseur de commandes pour extraire le nom et les arguments.
 * Gère les guillemets et les espaces dans les arguments.
 * 
 * @author Gabriel Chandesris
 * @version 1.0.0
 */
public class CommandParser {
    
    private static final Pattern QUOTED_ARG_PATTERN = Pattern.compile("\"([^\"]*)\"");
    private static final Pattern SIMPLE_ARG_PATTERN = Pattern.compile("[^\\s\\"]+");
    
    /**
     * Parse une ligne de commande complète.
     * 
     * @param input Ligne de commande
     * @return Tableau avec [nom_commande, arg1, arg2, ...]
     */
    public static String[] parse(String input) {
        if (input == null || input.trim().isEmpty()) {
            return new String[0];
        }
        
        String trimmed = input.trim();
        List<String> tokens = new ArrayList<>();
        
        // Utiliser un parseur manuel pour gérer les guillemets
        boolean inQuotes = false;
        StringBuilder current = new StringBuilder();
        
        for (int i = 0; i < trimmed.length(); i++) {
            char c = trimmed.charAt(i);
            
            if (c == '"') {
                inQuotes = !inQuotes;
            } else if (Character.isWhitespace(c) && !inQuotes) {
                if (current.length() > 0) {
                    tokens.add(current.toString());
                    current.setLength(0);
                }
            } else {
                current.append(c);
            }
        }
        
        if (current.length() > 0) {
            tokens.add(current.toString());
        }
        
        return tokens.toArray(new String[0]);
    }
    
    /**
     * Extrait le nom de la commande.
     * 
     * @param input Ligne de commande
     * @return Nom de la commande
     */
    public static String getCommandName(String input) {
        String[] parts = parse(input);
        if (parts.length > 0) {
            return parts[0];
        }
        return "";
    }
    
    /**
     * Extrait les arguments de la commande.
     * 
     * @param input Ligne de commande
     * @return Tableau d'arguments
     */
    public static String[] getArguments(String input) {
        String[] parts = parse(input);
        if (parts.length > 1) {
            String[] args = new String[parts.length - 1];
            System.arraycopy(parts, 1, args, 0, args.length);
            return args;
        }
        return new String[0];
    }
    
    /**
     * Vérifie si la commande a des arguments.
     * 
     * @param input Ligne de commande
     * @return True si des arguments sont présents
     */
    public static boolean hasArguments(String input) {
        return getArguments(input).length > 0;
    }
    
    /**
     * Nettoie un argument en supprimant les guillemets.
     * 
     * @param arg Argument à nettoyer
     * @return Argument nettoyé
     */
    public static String cleanArgument(String arg) {
        if (arg == null) return "";
        return arg.replaceAll("^\"|"$", "");
    }

// FICHIER : LsCommand.java
// DESCRIPTION : Commande 'ls' - Lister les fichiers et répertoires
// ============================================================================
package com.terminal.emulator.commands;

import com.terminal.emulator.TerminalState;
import com.terminal.emulator.filesystem.Directory;
import com.terminal.emulator.filesystem.FileNode;
import com.terminal.emulator.filesystem.TerminalFile;
import java.time.format.DateTimeFormatter;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

/**
 * Commande 'ls' pour lister le contenu d'un répertoire.
 * 
 * Options supportées:
 * - -l : Format long (détails)
 * - -a : Afficher les fichiers cachés
 * - -r : Inverser l'ordre
 * 
 * @author Gabriel Chandesris
 * @version 1.0.0
 */
public class LsCommand implements Command {
    
    private static final DateTimeFormatter DATE_FORMATTER = 
            DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm");
    
    /**
     * Exécute la commande ls.
     * 
     * @param state État du terminal
     * @param args Arguments (options et chemin)
     * @return Résultat de la commande
     */
    @Override
    public String execute(TerminalState state, String[] args) {
        boolean longFormat = false;
        boolean showHidden = false;
        boolean reverseOrder = false;
        String targetPath = ".";
        
        // Parser les arguments
        List<String> paths = new ArrayList<>();
        for (String arg : args) {
            if (arg.startsWith("-")) {
                if (arg.contains("l")) longFormat = true;
                if (arg.contains("a")) showHidden = true;
                if (arg.contains("r")) reverseOrder = true;
            } else if (!arg.isEmpty()) {
                paths.add(arg);
            }
        }
        
        // Si des chemins sont spécifiés, utiliser le premier
        if (!paths.isEmpty()) {
            targetPath = paths.get(0);
        }
        
        // Résoudre le chemin
        Directory targetDir = resolveDirectory(state, targetPath);
        if (targetDir == null) {
            return "ls: cannot access '" + targetPath + "': No such file or directory";
        }
        
        // Récupérer les enfants
        List<FileNode> children = new ArrayList<>(targetDir.getChildren());
        
        // Filtrer les fichiers cachés si nécessaire
        if (!showHidden) {
            children.removeIf(node -> node.getName().startsWith("."));
        }
        
        // Trier
        Collections.sort(children, (n1, n2) -> {
            int result = n1.getName().compareTo(n2.getName());
            return reverseOrder ? -result : result;
        });
        
        // Formater la sortie
        StringBuilder output = new StringBuilder();
        
        if (longFormat) {
            output.append("total ").append(children.size()).append("\n");
            for (FileNode node : children) {
                output.append(formatLong(node));
            }
        } else {
            for (int i = 0; i < children.size(); i++) {
                if (i > 0) output.append("  ");
                output.append(children.get(i).getName());
            }
            if (!children.isEmpty()) output.append("\n");
        }
        
        return output.toString();
    }
    
    /**
     * Résout un chemin en répertoire.
     * 
     * @param state État du terminal
     * @param path Chemin à résoudre
     * @return Répertoire ou null
     */
    private Directory resolveDirectory(TerminalState state, String path) {
        if (".".equals(path) || "".equals(path)) {
            return state.getCurrentDirectory();
        }
        
        if ("/".equals(path)) {
            return state.getRootDirectory();
        }
        
        if (path.startsWith("/")) {
            // Chemin absolu
            return resolveAbsolutePath(state.getRootDirectory(), path);
        } else {
            // Chemin relatif
            return resolveRelativePath(state.getCurrentDirectory(), path);
        }
    }
    
    /**
     * Résout un chemin absolu.
     */
    private Directory resolveAbsolutePath(Directory root, String path) {
        String[] parts = path.split("/");
        Directory current = root;
        
        for (int i = 1; i < parts.length; i++) {
            if (parts[i].isEmpty() || ".".equals(parts[i])) continue;
            if ("..".equals(parts[i])) {
                if (current.getParent() != null) {
                    current = current.getParent();
                }
            } else {
                FileNode node = current.getChild(parts[i]);
                if (node == null || !node.isDirectory()) {
                    return null;
                }
                current = (Directory) node;
            }
        }
        
        return current;
    }
    
    /**
     * Résout un chemin relatif.
     */
    private Directory resolveRelativePath(Directory current, String path) {
        String[] parts = path.split("/");
        
        for (String part : parts) {
            if (part.isEmpty() || ".".equals(part)) continue;
            if ("..".equals(part)) {
                if (current.getParent() != null) {
                    current = current.getParent();
                }
            } else {
                FileNode node = current.getChild(part);
                if (node == null || !node.isDirectory()) {
                    return null;
                }
                current = (Directory) node;
            }
        }
        
        return current;
    }
    
    /**
     * Formate un nœud en format long.
     */
    private String formatLong(FileNode node) {
        String type = node.isDirectory() ? "d" : "-";
        String permissions = node.isDirectory() ? "rwxr-xr-x" : "rw-r--r--";
        String size = node.isDirectory() ? "4096" : String.valueOf(node.getSize());
        String date = node.getModifiedAt().format(DATE_FORMATTER);
        
        return String.format("%s %s 1 user group %8s %s %s\n", 
                type, permissions, size, date, node.getName());
    }
    
    /**
     * @return Le nom de la commande
     */
    @Override
    public String getName() {
        return "ls";
    }
    
    /**
     * @return La description de la commande
     */
    @Override
    public String getDescription() {
        return "List directory contents";
    }
    
    /**
     * @return L'usage de la commande
     */
    @Override
    public String getUsage() {
        return "ls [OPTION]... [FILE]...";
    }
}

    /**
     * Exécute la commande cat.
     * 
     * @param state État du terminal
     * @param args Arguments (noms de fichiers)
     * @return Contenu des fichiers ou message d'erreur
     */
    @Override
    public String execute(TerminalState state, String[] args) {
        if (args.length == 0) {
            return "cat: missing operand";
        }
        
        StringBuilder output = new StringBuilder();
        
        for (String fileName : args) {
            FileNode node = resolveFile(state, fileName);

                continue;
            }
            
            TerminalFile file = (TerminalFile) node;
            output.append(file.getContent());
        }
        
        return output.toString();
    }

                    current = current.getParent();
                }
            } else {
                FileNode node = current.getChild(parts[i]);
                if (node == null) {
                    return null;
                }
                if (i == parts.length - 1) {
                    return node;
                }
                if (!node.isDirectory()) {
                    return null;
                }
                current = (Directory) node;
            }
        }
        
        return current;
    }
    
    /**
     * Résout un chemin relatif.
     */
    private FileNode resolveRelativePath(Directory current, String path) {
        String[] parts = path.split("/");
        
        for (int i = 0; i < parts.length; i++) {

/**
 * Commande 'touch' pour créer un fichier vide ou mettre à jour sa date de modification.
 * 
 * @author Gabriel Chandesris
 * @version 1.0.0
 */
public class TouchCommand implements Command {
    
    /**
     * Exécute la commande touch.
     * 
     * @param state État du terminal
     * @param args Arguments (noms de fichiers)
     * @return Message de résultat
     */
    @Override
    public String execute(TerminalState state, String[] args) {
        if (args.length == 0) {
            return "touch: missing file operand";
        }
        
        StringBuilder output = new StringBuilder();
        
        for (String fileName : args) {
            FileNode existing = resolveFile(state, fileName);
            
            if (existing != null && existing.isDirectory()) {
                output.append("touch: cannot touch '").append(fileName).append("': Is a directory\n");
                continue;
            }
            
            Directory parentDir = getParentDirectory(state, fileName);
            if (parentDir == null) {
                output.append("touch: cannot touch '").append(fileName).append("': No such file or directory\n");
                continue;
            }
            
            String simpleName = getSimpleName(fileName);
            
            if (existing != null && existing.isFile()) {
                // Mettre à jour la date de modification
                ((TerminalFile) existing).setContent(existing.getContent());
            } else {
                // Créer un nouveau fichier
                parentDir.createFile(simpleName);
            }
        }
        
        return output.toString();
    }

     * Résout un nom de fichier en nœud FileNode.

    private FileNode resolveFile(TerminalState state, String fileName) {

                   "Try 'cp --help' for more information.";
        }
        
        // Cas simple : cp source destination
        if (args.length == 2) {
            return copyFile(state, args[0], args[1]);
        }
        
        // Cas multiple : cp source1 source2 ... destination_directory
        String destination = args[args.length - 1];

     */

        FileNode source = resolveFile(state, sourcePath);

            return "cp: cannot stat '" + sourcePath + "': No such file or directory";
        }

            ((TerminalFile) existing).setContent(sourceFile.getContent());
        } else {
            // Créer un nouveau fichier
            TerminalFile sourceFile = (TerminalFile) source;
            TerminalFile newFile = new TerminalFile(destName, destParent, sourceFile.getContent());

    /**

    private FileNode resolveRelativePath(Directory current, String path) {
        String[] parts = path.split("/");
        
        for (int i = 0; i < parts.length; i++) {
            if (parts[i].isEmpty() || ".".equals(parts[i])) continue;
            if ("..".equals(parts[i])) {
                if (current.getParent() != null) {
                    current = current.getParent();
                }
            } else {
                FileNode node = current.getChild(parts[i]);
                if (node == null) {
                    return null;
                }
                if (i == parts.length - 1) {
                    return node;
                }
                if (!node.isDirectory()) {
                    return null;
                }
                current = (Directory) node;
            }
        }
        
        return current;
    }

package com.terminal.emulator.commands;

import com.terminal.emulator.TerminalState;
import java.util.Map;

/**
 * Commande 'help' pour afficher l'aide sur les commandes disponibles.
 * 
 * @author Gabriel Chandesris
 * @version 1.0.0
 */
public class HelpCommand implements Command {
    
    /**
     * Exécute la commande help.
     * 
     * @param state État du terminal
     * @param args Arguments (nom de commande spécifique ou vide pour l'aide générale)
     * @return Texte d'aide
     */
    @Override
    public String execute(TerminalState state, String[] args) {
        if (args.length == 0) {
            return getGeneralHelp();
        } else {
            return getCommandHelp(args[0]);
        }
    }
    
    /**
     * Retourne l'aide générale.
     */
    private String getGeneralHelp() {
        StringBuilder help = new StringBuilder();
        help.append("Available commands:\n\n");
        
        Map<String, Command> commands = CommandFactory.getAllCommands();
        for (Map.Entry<String, Command> entry : commands.entrySet()) {
            Command cmd = entry.getValue();
            help.append(String.format("  %-15s %s\n", cmd.getName(), cmd.getDescription()));
        }
        
        help.append("\nType 'help <command>' for more information about a specific command.\n");
        return help.toString();
    }
    
    /**
     * Retourne l'aide pour une commande spécifique.
     */
    private String getCommandHelp(String commandName) {
        Command cmd = CommandFactory.getCommand(commandName);
        
        if (cmd == null) {
            return "help: no help topics match '" + commandName + "'.\n" +
                   "Try 'help' for a list of available commands.";
        }
        
        return String.format("Usage: %s\n\n%s\n", cmd.getUsage(), cmd.getDescription());
    }
    
    /**
     * @return Le nom de la commande
     */
    @Override
    public String getName() {
        return "help";
    }
    
    /**
     * @return La description de la commande
     */
    @Override
    public String getDescription() {
        return "Display help information";
    }
    
    /**
     * @return L'usage de la commande
     */
    @Override
    public String getUsage() {
        return "help [COMMAND]";
    }
}

        promptLabel.setText(state.getPrompt() + " ");

    /**
     * Écouteur pour l'exécution des commandes.
     */
    private class CommandActionListener implements ActionListener {
        @Override
        public void actionPerformed(ActionEvent e) {
            String command = inputField.getText().trim();
            
            if (command.isEmpty()) {
                clearInputLine();
                return;
            }
            
            // Ajouter à l'historique
            state.addToHistory(command);
            state.resetHistoryIndex();
            
            // Afficher la commande dans la sortie
            appendOutput(state.getPrompt() + " " + command + "\n");
            
            // Parser et exécuter la commande
            String[] parts = CommandParser.parse(command);
            if (parts.length == 0) {
                clearInputLine();
                return;
            }
            
            String commandName = parts[0];
            String[] args = new String[parts.length - 1];
            System.arraycopy(parts, 1, args, 0, args.length);
            
            Command cmd = CommandFactory.getCommand(commandName);
            
            if (cmd == null) {
                appendOutput(commandName + ": command not found\n");
            } else {
                String result = cmd.execute(state, args);
                
                // Gestion des codes spéciaux
                if ("EXIT".equals(result)) {
                    System.exit(0);
                } else if (result.contains("\033[H\033[2J")) {
                    // Code ANSI pour clear
                    clearScreen();
                    printWelcomeMessage();
                } else {
                    appendOutput(result + "\n");
                }
            }
            
            // Mettre à jour le prompt et effacer la saisie
            updatePrompt();
            clearInputLine();
        }
    }
    
    /**
     * Écouteur pour la navigation dans l'historique des commandes.
     */
    private class HistoryKeyListener extends KeyAdapter {
        @Override
        public void keyPressed(KeyEvent e) {
            if (e.getKeyCode() == KeyEvent.VK_UP) {
                // Flèche haut : commande précédente
                String prevCommand = state.getPreviousCommand();
                if (prevCommand != null) {
                    inputField.setText(prevCommand);
                }
                e.consume();
            } else if (e.getKeyCode() == KeyEvent.VK_DOWN) {
                // Flèche bas : commande suivante
                String nextCommand = state.getNextCommand();
                if (nextCommand != null) {
                    inputField.setText(nextCommand);
                }
                e.consume();
            } else if (e.getKeyCode() == KeyEvent.VK_TAB) {
                // Tab : auto-complétion (simplifiée)
                handleTabCompletion();
                e.consume();
            }
        }
        
        /**
         * Gère l'auto-complétion avec la touche Tab.
         */
        private void handleTabCompletion() {
            String text = inputField.getText();
            String[] parts = CommandParser.parse(text);
            
            if (parts.length == 0) {
                return;
            }
            
            // Complétion des commandes
            if (parts.length == 1) {
                String prefix = parts[0];
                String[] commandNames = CommandFactory.getCommandNames();
                
                for (String cmdName : commandNames) {
                    if (cmdName.startsWith(prefix)) {
                        inputField.setText(cmdName + " ");
                        return;
                    }
                }
            }
        }
    }
    
    /**
     * @return L'état du terminal
     */
    public TerminalState getState() {
        return state;
    }
    
    /**
     * Définit l'état du terminal.
     * 
     * @param state Nouveau état
     */
    public void setState(TerminalState state) {
        this.state = state;
        updatePrompt();
    }

    /**
     * Exécute un script à partir d'un fichier.
     * 
     * @param file Fichier contenant le script
     * @return Résultat de l'exécution
     */
    public String executeFile(TerminalFile file) {
        if (file == null) {
            return "Script file not found";
        }
        return execute(file.getContent());
    }
    
    /**
     * Exécute un script à partir d'une chaîne de caractères.
     * 
     * @param script Contenu du script
     * @return Résultat de l'exécution
     */
    public String execute(String script) {
        if (script == null || script.trim().isEmpty()) {
            return "";
        }
        
        StringBuilder output = new StringBuilder();
        String[] lines = script.split("\n");
        
        for (String line : lines) {
            String trimmed = line.trim();
            
            // Ignorer les commentaires
            if (trimmed.startsWith("#") || trimmed.isEmpty()) {
                continue;
            }
            
            // Exécuter la ligne
            String result = executeLine(trimmed);
            if (result != null && !result.isEmpty()) {
                output.append(result).append("\n");
            }
        }
        
        return output.toString();
    }
    
    /**
     * Exécute une seule ligne de script.
     * 
     * @param line Ligne à exécuter
     * @return Résultat de l'exécution
     */
    private String executeLine(String line) {
        // Vérifier les structures de contrôle
        if (line.startsWith("if ")) {
            return executeIf(line);
        } else if (line.startsWith("while ")) {
            return executeWhile(line);
        } else if (line.startsWith("for ")) {
            return executeFor(line);
        } else if (line.equals("fi") || line.equals("done") || line.equals("end")) {
            return null; // Fin de bloc, ne rien retourner
        } else if (line.startsWith("set ")) {
            return executeSet(line);
        } else if (line.startsWith("echo ")) {
            return executeScriptEcho(line);
        } else if (line.startsWith("cd ")) {
            return executeScriptCd(line);
        } else if (line.startsWith("pwd")) {
            return state.getCurrentDirectory().getPath();
        } else if (line.startsWith("ls")) {
            return executeLs(line);
        } else if (line.startsWith("cat")) {
            return executeCat(line);
        }
        
        // Exécuter comme commande normale
        String[] parts = CommandParser.parse(line);
        if (parts.length > 0) {
            Command cmd = CommandFactory.getCommand(parts[0]);
            if (cmd != null) {
                String[] args = new String[parts.length - 1];
                System.arraycopy(parts, 1, args, 0, args.length);
                return cmd.execute(state, args);
            }
        }
        
        return "Script error: unknown command '" + line.split(" ")[0] + "'";
    }
    
    /**
     * Exécute une instruction if.
     */
    private String executeIf(String line) {
        // Format: if [ condition ]
        String condition = line.substring(3, line.length() - 1).trim();
        boolean result = evaluateCondition(condition);
        
        context.setInIfBlock(true);
        context.setIfConditionResult(result);
        context.setIfBlockDepth(1);
        
        return null;
    }
    
    /**
     * Exécute une boucle while.
     */
    private String executeWhile(String line) {
        // Format: while [ condition ]
        String condition = line.substring(6, line.length() - 1).trim();
        boolean result = evaluateCondition(condition);
        
        context.setInWhileBlock(true);
        context.setWhileCondition(condition);
        context.setWhileConditionResult(result);
        context.setWhileBlockDepth(1);
        
        return null;
    }
    
    /**
     * Exécute une boucle for.
     */
    private String executeFor(String line) {
        // Format: for var in list
        String[] parts = line.substring(4).trim().split(" in ");
        if (parts.length != 2) {
            return "Script error: invalid for syntax";
        }
        
        String varName = parts[0].trim();
        String[] values = parts[1].trim().split(" ");
        
        context.setInForBlock(true);
        context.setForVariable(varName);
        context.setForValues(values);
        context.setForIndex(0);
        context.setForBlockDepth(1);
        
        // Définir la première valeur
        if (values.length > 0) {
            context.getVariables().put(varName, values[0]);
        }
        
        return null;
    }
    
    /**
     * Exécute une instruction set (définition de variable).
     */
    private String executeSet(String line) {
        // Format: set var=value
        String[] parts = line.substring(4).trim().split("=", 2);
        if (parts.length != 2) {
            return "Script error: invalid set syntax";
        }
        
        String varName = parts[0].trim();
        String value = parts[1].trim();
        
        // Remplacer les références de variables
        value = replaceVariables(value);
        
        context.getVariables().put(varName, value);
        return null;
    }
    
    /**
     * Exécute une instruction echo dans un script.
     */
    private String executeScriptEcho(String line) {
        // Format: echo text
        String text = line.substring(5).trim();
        return replaceVariables(text);
    }
    
    /**
     * Exécute une instruction cd dans un script.
     */
    private String executeScriptCd(String line) {
        String path = line.substring(3).trim();
        
        // Résoudre le chemin
        Directory newDir = resolveDirectory(state, path);
        if (newDir == null) {
            return "cd: no such file or directory: " + path;
        }
        
        state.setCurrentDirectory(newDir);
        return null;
    }
    
    /**
     * Exécute une instruction ls dans un script.
     */
    private String executeLs(String line) {
        String[] parts = CommandParser.parse(line);
        String[] args = new String[parts.length - 1];
        System.arraycopy(parts, 1, args, 0, args.length);
        
        Command cmd = CommandFactory.getCommand("ls");
        if (cmd != null) {
            return cmd.execute(state, args);
        }
        return "ls: command not available";
    }
    
    /**
     * Exécute une instruction cat dans un script.
     */
    private String executeCat(String line) {
        String[] parts = CommandParser.parse(line);
        String[] args = new String[parts.length - 1];
        System.arraycopy(parts, 1, args, 0, args.length);
        
        Command cmd = CommandFactory.getCommand("cat");
        if (cmd != null) {
            return cmd.execute(state, args);
        }
        return "cat: command not available";
    }
    
    /**
     * Évalue une condition.
     */
    private boolean evaluateCondition(String condition) {
        // Supporte: [ "$var" = "value" ], [ -f "file" ], [ -d "dir" ]
        condition = condition.trim();
        
        if (condition.startsWith("$")) {
            // Comparaison de variable
            String[] parts = condition.split("=");
            if (parts.length == 2) {
                String varName = parts[0].trim();
                String value = parts[1].trim().replaceAll("^\"|"$", "");
                
                String varValue = context.getVariables().getOrDefault(varName, "");
                return varValue.equals(value);
            }
        } else if (condition.startsWith("-")) {
            // Test de fichier
            String[] parts = condition.split(" ");
            if (parts.length == 2) {
                String test = parts[0];
                String path = parts[1].trim().replaceAll("^\"|"$", "");
                
                FileNode node = resolveFile(state, path);
                
                if ("-f".equals(test)) {
                    return node != null && node.isFile();
                } else if ("-d".equals(test)) {
                    return node != null && node.isDirectory();
                } else if ("-e".equals(test)) {
                    return node != null;
                }
            }
        }
        
        return false;
    }
    
    /**
     * Remplace les références de variables dans une chaîne.
     */
    private String replaceVariables(String text) {
        String result = text;
        
        for (Map.Entry<String, String> entry : context.getVariables().entrySet()) {
            result = result.replace("$" + entry.getKey(), entry.getValue());
        }
        
        return result;
    }
    
    /**
     * Résout un chemin en FileNode.
     */
    private FileNode resolveFile(TerminalState state, String fileName) {
        Directory current = state.getCurrentDirectory();
        
        if (fileName.startsWith("/")) {
            return resolveAbsolutePath(state.getRootDirectory(), fileName);
        } else {
            return resolveRelativePath(current, fileName);
        }
    }
    
    /**
     * Résout un chemin en Directory.
     */
    private Directory resolveDirectory(TerminalState state, String dirName) {
        Directory current = state.getCurrentDirectory();
        
        if (dirName.startsWith("/")) {
            return resolveAbsoluteDirectory(state.getRootDirectory(), dirName);
        } else {
            return resolveRelativeDirectory(current, dirName);
        }
    }
    
    /**
     * Résout un chemin absolu en FileNode.
     */
    private FileNode resolveAbsolutePath(Directory root, String path) {
        String[] parts = path.split("/");
        Directory current = root;
        
        for (int i = 1; i < parts.length; i++) {
            if (parts[i].isEmpty() || ".".equals(parts[i])) continue;
            if ("..".equals(parts[i])) {
                if (current.getParent() != null) {
                    current = current.getParent();
                }
            } else {
                FileNode node = current.getChild(parts[i]);
                if (node == null) {
                    return null;
                }
                if (i == parts.length - 1) {
                    return node;
                }
                if (!node.isDirectory()) {
                    return null;
                }
                current = (Directory) node;
            }
        }
        
        return current;
    }
    
    /**
     * Résout un chemin relatif en FileNode.
     */
    private FileNode resolveRelativePath(Directory current, String path) {
        String[] parts = path.split("/");
        
        for (int i = 0; i < parts.length; i++) {
            if (parts[i].isEmpty() || ".".equals(parts[i])) continue;
            if ("..".equals(parts[i])) {
                if (current.getParent() != null) {
                    current = current.getParent();
                }
            } else {
                FileNode node = current.getChild(parts[i]);
                if (node == null) {
                    return null;
                }
                if (i == parts.length - 1) {
                    return node;
                }
                if (!node.isDirectory()) {
                    return null;
                }
                current = (Directory) node;
            }
        }
        
        return current;
    }
    
    /**
     * Résout un chemin absolu en Directory.
     */
    private Directory resolveAbsoluteDirectory(Directory root, String path) {
        String[] parts = path.split("/");
        Directory current = root;
        
        for (int i = 1; i < parts.length; i++) {
            if (parts[i].isEmpty() || ".".equals(parts[i])) continue;
            if ("..".equals(parts[i])) {
                if (current.getParent() != null) {
                    current = current.getParent();
                }
            } else {
                FileNode node = current.getChild(parts[i]);
                if (node == null || !node.isDirectory()) {
                    return null;
                }
                current = (Directory) node;
            }
        }
        
        return current;
    }
    
    /**
     * Résout un chemin relatif en Directory.
     */
    private Directory resolveRelativeDirectory(Directory current, String path) {
        String[] parts = path.split("/");
        
        for (String part : parts) {
            if (part.isEmpty() || ".".equals(part)) continue;
            if ("..".equals(part)) {
                if (current.getParent() != null) {
                    current = current.getParent();
                }
            } else {
                FileNode node = current.getChild(part);
                if (node == null || !node.isDirectory()) {
                    return null;
                }
                current = (Directory) node;
            }
        }
        
        return current;
    }
    
    /**
     * @return Le contexte du script
     */
    public ScriptContext getContext() {
        return context;
    }
    
    /**
     * @return L'état du terminal
     */
    public TerminalState getState() {
        return state;
    }
    
    /**
     * Définit l'état du terminal.
     * 
     * @param state Nouveau état
     */
    public void setState(TerminalState state) {
        this.state = state;
    }
}

package com.terminal.emulator.script;

import java.util.ArrayList;
import java.util.List;

/**
 * Parseur de scripts pour l'analyse syntaxique.
 * 
 * @author Gabriel Chandesris
 * @version 1.0.0
 */
public class ScriptParser {
    
    /**
     * Parse un script et retourne les blocs de code.
     * 
     * @param script Contenu du script
     * @return Liste de blocs de code
     */
    public static List<ScriptBlock> parse(String script) {
        List<ScriptBlock> blocks = new ArrayList<>();
        
        if (script == null || script.trim().isEmpty()) {
            return blocks;
        }
        
        String[] lines = script.split("\n");
        ScriptBlock currentBlock = new ScriptBlock(ScriptBlock.Type.NORMAL, 0);
        int depth = 0;
        
        for (int i = 0; i < lines.length; i++) {
            String line = lines[i].trim();
            
            // Ignorer les commentaires et lignes vides
            if (line.isEmpty() || line.startsWith("#")) {
                if (currentBlock.getType() != ScriptBlock.Type.NORMAL) {
                    currentBlock.addLine(line);
                }
                continue;
            }
            
            // Détecter les blocs de contrôle
            if (line.startsWith("if ")) {
                if (currentBlock.getType() != ScriptBlock.Type.NORMAL) {
                    blocks.add(currentBlock);
                }
                currentBlock = new ScriptBlock(ScriptBlock.Type.IF, i);
                currentBlock.addLine(line);
                depth++;
            } else if (line.startsWith("while ")) {
                if (currentBlock.getType() != ScriptBlock.Type.NORMAL) {
                    blocks.add(currentBlock);
                }
                currentBlock = new ScriptBlock(ScriptBlock.Type.WHILE, i);
                currentBlock.addLine(line);
                depth++;
            } else if (line.startsWith("for ")) {
                if (currentBlock.getType() != ScriptBlock.Type.NORMAL) {
                    blocks.add(currentBlock);
                }
                currentBlock = new ScriptBlock(ScriptBlock.Type.FOR, i);
                currentBlock.addLine(line);
                depth++;
            } else if (line.equals("fi") || line.equals("done") || line.equals("end")) {
                currentBlock.addLine(line);
                blocks.add(currentBlock);
                depth--;
                if (depth >= 0) {
                    currentBlock = new ScriptBlock(ScriptBlock.Type.NORMAL, i);
                }
            } else {
                currentBlock.addLine(line);
            }
        }
        
        if (!currentBlock.getLines().isEmpty()) {
            blocks.add(currentBlock);
        }
        
        return blocks;
    }
    
    /**
     * Vérifie la syntaxe d'un script.
     * 
     * @param script Contenu du script
     * @return True si la syntaxe est valide
     */
    public static boolean validateSyntax(String script) {
        if (script == null) {
            return false;
        }
        
        String[] lines = script.split("\n");
        int ifDepth = 0;
        int whileDepth = 0;
        int forDepth = 0;
        
        for (String line : lines) {
            String trimmed = line.trim();
            
            if (trimmed.startsWith("if ")) {
                ifDepth++;
            } else if (trimmed.equals("fi")) {
                ifDepth--;
            } else if (trimmed.startsWith("while ")) {
                whileDepth++;
            } else if (trimmed.equals("done")) {
                // 'done' peut terminer un while ou un for
                if (whileDepth > 0) {
                    whileDepth--;
                } else if (forDepth > 0) {
                    forDepth--;
                }
            } else if (trimmed.startsWith("for ")) {
                forDepth++;
            } else if (trimmed.equals("end")) {
                forDepth--;
            }
        }
        
        return ifDepth == 0 && whileDepth == 0 && forDepth == 0;
    }
    
    /**
     * Classe représentant un bloc de script.
     */
    public static class ScriptBlock {
        
        public enum Type {
            NORMAL, IF, WHILE, FOR
        }
        
        private Type type;
        private int startLine;
        private List<String> lines;
        
        /**
         * Constructeur.
         * 
         * @param type Type du bloc
         * @param startLine Ligne de début
         */
        public ScriptBlock(Type type, int startLine) {
            this.type = type;
            this.startLine = startLine;
            this.lines = new ArrayList<>();
        }
        
        /**
         * Ajoute une ligne au bloc.
         * 
         * @param line Ligne à ajouter
         */
        public void addLine(String line) {
            lines.add(line);
        }
        
        /**
         * @return Le type du bloc
         */
        public Type getType() {
            return type;
        }
        
        /**
         * @return La ligne de début
         */
        public int getStartLine() {
            return startLine;
        }
        
        /**
         * @return La liste des lignes
         */
        public List<String> getLines() {
            return lines;
        }
        
        /**
         * @return Le contenu du bloc
         */
        public String getContent() {
            return String.join("\n", lines);
        }
        
        /**
         * @return Une représentation textuelle
         */
        @Override
        public String toString() {
            return "ScriptBlock{" +
                    "type=" + type +
                    ", startLine=" + startLine +
                    ", lines=" + lines +
                    '}';
        }
    }
}

import com.terminal.emulator.TerminalState;
import com.terminal.emulator.commands.*;
import com.terminal.emulator.filesystem.Directory;
import com.terminal.emulator.filesystem.TerminalFile;
import org.junit.jupiter.api.*;
import static org.junit.jupiter.api.Assertions.*;

/**
 * Tests d'intégration pour l'émulateur de terminal complet.
 * Teste l'interaction entre les différents composants.
 * 
 * @author Gabriel Chandesris
 * @version 1.0.0
 */
class IntegrationTest {
    
    private TerminalState state;
    private Directory root;
    private Directory home;
    
    @BeforeEach
    void setUp() {
        state = new TerminalState();
        root = state.getRootDirectory();
        home = root.createDirectory("home");
        state.setCurrentDirectory(home);
        
        // Créer une structure de fichiers de test
        Directory docs = home.createDirectory("documents");
        Directory images = home.createDirectory("images");
        
        TerminalFile readme = home.createFile("README.md");
        readme.setContent("# Welcome to the Terminal Emulator");
        
        TerminalFile file1 = docs.createFile("file1.txt");
        file1.setContent("Content of file 1");
        
        TerminalFile file2 = docs.createFile("file2.txt");
        file2.setContent("Content of file 2");
    }
    
    @AfterEach
    void tearDown() {
        state = null;
        root = null;
        home = null;
    }
    
    @Test
    void testCompleteWorkflow() {
        // 1. Lister le répertoire home
        Command ls = CommandFactory.getCommand("ls");
        String result = ls.execute(state, new String[0]);
        assertTrue(result.contains("README.md"));
        assertTrue(result.contains("documents"));
        assertTrue(result.contains("images"));
        
        // 2. Changer de répertoire
        Command cd = CommandFactory.getCommand("cd");
        result = cd.execute(state, new String[]{"documents"});
        assertEquals("", result);
        assertEquals("/home/documents", state.getCurrentDirectory().getPath());
        
        // 3. Lister le répertoire documents
        result = ls.execute(state, new String[0]);
        assertTrue(result.contains("file1.txt"));
        assertTrue(result.contains("file2.txt"));
        
        // 4. Afficher le contenu d'un fichier
        Command cat = CommandFactory.getCommand("cat");
        result = cat.execute(state, new String[]{"file1.txt"});
        assertEquals("Content of file 1", result);
        
        // 5. Retour au répertoire parent
        result = cd.execute(state, new String[]{".."});
        assertEquals("", result);
        assertEquals("/home", state.getCurrentDirectory().getPath());
    }
    
    @Test
    void testFileOperationsWorkflow() {
        // Créer un fichier
        Command touch = CommandFactory.getCommand("touch");
        String result = touch.execute(state, new String[]{"newfile.txt"});
        assertEquals("", result);
        
        TerminalFile newFile = (TerminalFile) home.getChild("newfile.txt");
        assertNotNull(newFile);
        
        // Écrire dans le fichier via echo (simulation)
        Command echo = CommandFactory.getCommand("echo");
        result = echo.execute(state, new String[]{"Hello"});
        assertEquals("Hello", result);
        
        // Copier le fichier
        Command cp = CommandFactory.getCommand("cp");
        result = cp.execute(state, new String[]{"newfile.txt", "newfile_copy.txt"});
        assertEquals("", result);
        
        TerminalFile copy = (TerminalFile) home.getChild("newfile_copy.txt");
        assertNotNull(copy);
        
        // Déplacer le fichier
        Command mv = CommandFactory.getCommand("mv");
        result = mv.execute(state, new String[]{"newfile.txt", "moved.txt"});
        assertEquals("", result);
        
        assertNull(home.getChild("newfile.txt"));
        assertNotNull(home.getChild("moved.txt"));
        
        // Supprimer le fichier
        Command rm = CommandFactory.getCommand("rm");
        result = rm.execute(state, new String[]{"moved.txt"});
        assertEquals("", result);
        
        assertNull(home.getChild("moved.txt"));
    }
    
    @Test
    void testDirectoryOperationsWorkflow() {
        // Créer un répertoire
        Command mkdir = CommandFactory.getCommand("mkdir");
        String result = mkdir.execute(state, new String[]{"newdir"});
        assertEquals("", result);
        
        Directory newDir = (Directory) home.getChild("newdir");
        assertNotNull(newDir);
        
        // Changer vers le nouveau répertoire
        Command cd = CommandFactory.getCommand("cd");
        result = cd.execute(state, new String[]{"newdir"});
        assertEquals("", result);
        assertEquals("/home/newdir", state.getCurrentDirectory().getPath());
        
        // Créer un fichier dans le nouveau répertoire
        Command touch = CommandFactory.getCommand("touch");
        result = touch.execute(state, new String[]{"test.txt"});
        assertEquals("", result);
        
        // Retour au répertoire parent
        result = cd.execute(state, new String[]{".."});
        assertEquals("", result);
        assertEquals("/home", state.getCurrentDirectory().getPath());
        
        // Supprimer le répertoire (doit être vide)
        Command rmdir = CommandFactory.getCommand("rmdir");
        result = rmdir.execute(state, new String[]{"newdir"});
        assertEquals("", result);
        
        assertNull(home.getChild("newdir"));
    }
    
    @Test
    void testCommandHistoryIntegration() {
        // Exécuter plusieurs commandes
        Command ls = CommandFactory.getCommand("ls");
        ls.execute(state, new String[0]);
        
        Command cd = CommandFactory.getCommand("cd");
        cd.execute(state, new String[]{"documents"});
        
        Command pwd = CommandFactory.getCommand("pwd");
        pwd.execute(state, new String[0]);
        
        // Vérifier l'historique
        assertEquals(3, state.getCommandHistory().size());
        assertEquals("ls", state.getCommandHistory().get(0));
        assertEquals("cd documents", state.getCommandHistory().get(1));
        assertEquals("pwd", state.getCommandHistory().get(2));
        
        // Navigation dans l'historique
        state.resetHistoryIndex();
        String prev = state.getPreviousCommand();
        assertEquals("pwd", prev);
        
        prev = state.getPreviousCommand();
        assertEquals("cd documents", prev);
    }
    
    @Test
    void testErrorHandling() {
        // Commande non existante
        Command ls = CommandFactory.getCommand("ls");
        String result = ls.execute(state, new String[]{"nonexistent"});
        assertTrue(result.contains("No such file or directory"));
        
        // Changer vers un répertoire non existant
        Command cd = CommandFactory.getCommand("cd");
        result = cd.execute(state, new String[]{"nonexistent"});
        assertTrue(result.contains("no such file or directory"));
        
        // Supprimer un fichier non existant
        Command rm = CommandFactory.getCommand("rm");
        result = rm.execute(state, new String[]{"nonexistent.txt"});
        assertTrue(result.contains("No such file or directory"));
    }
    
    @Test
    void testPathResolution() {
        // Chemin absolu
        Command cd = CommandFactory.getCommand("cd");
        String result = cd.execute(state, new String[]{"/home/documents"});
        assertEquals("", result);
        assertEquals("/home/documents", state.getCurrentDirectory().getPath());
        
        // Chemin relatif
        result = cd.execute(state, new String[]{".."});
        assertEquals("", result);
        assertEquals("/home", state.getCurrentDirectory().getPath());
        
        // Chemin avec ..
        result = cd.execute(state, new String[]{"documents/../images"});
        assertEquals("", result);
        assertEquals("/home/images", state.getCurrentDirectory().getPath());
    }
}

/*
 * DOCUMENTATION TECHNIQUE COMPLÈTE
 * ==================================
 * 
 * 1. ARCHITECTURE DU PROJET
 * -------------------------
 * 
 * Le projet est organisé en plusieurs packages principaux :
 * 
 * - com.terminal.emulator : Package racine contenant la classe Main et TerminalState
 * - com.terminal.emulator.gui : Interface graphique Swing
 * - com.terminal.emulator.filesystem : Système de fichiers virtuel
 * - com.terminal.emulator.commands : Commandes du terminal
 * - com.terminal.emulator.script : Moteur de scripts minimaliste
 * - tests : Tests unitaires et d'intégration
 * 
 * 2. SYSTÈME DE FICHIERS VIRTUEL
 * ------------------------------
 * 
 * Le système de fichiers est implémenté avec une structure arborescente :
 * 
 * - FileNode : Classe abstraite de base pour tous les nœuds
 * - Directory : Répertoire, peut contenir d'autres FileNode
 * - TerminalFile : Fichier texte avec contenu
 * 
 * Chaque nœud a :
 * - Un nom
 * - Un parent (sauf la racine)
 * - Une date de création
 * - Une date de modification
 * - Un chemin absolu
 * 
 * 3. COMMANDES IMPLÉMENTÉES
 * -------------------------
 * 
 * Commandes de base :
 * - ls : Lister les fichiers (options : -l, -a, -r)
 * - cd : Changer de répertoire
 * - pwd : Afficher le répertoire courant
 * - cat : Afficher le contenu d'un fichier
 * - echo : Afficher du texte
 * - touch : Créer un fichier ou mettre à jour sa date
 * - mkdir : Créer un répertoire
 * - rm : Supprimer un fichier
 * - rmdir : Supprimer un répertoire vide
 * - cp : Copier un fichier
 * - mv : Déplacer/renommer un fichier
 * - clear : Effacer l'écran
 * - help : Afficher l'aide
 * - exit : Quitter le terminal
 * 
 * 4. MOTEUR DE SCRIPTS MINIMALISTE
 * --------------------------------
 * 
 * Le langage de script supporte :
 * 
 * - Variables : set var=value, $var
 * - Conditions : if [ condition ]; fi
 *   * Comparaison : [ "$var" = "value" ]
 *   * Tests de fichiers : [ -f file ], [ -d dir ], [ -e path ]
 * - Boucles : while [ condition ]; done
 * - Boucles for : for var in list; do ... done
 * - Commentaires : # commentaire
 * - Commandes : toutes les commandes du terminal
 * 
 * 5. INTERFACE GRAPHIQUE
 * ----------------------
 * 
 * Composants principaux :
 * - TerminalFrame : Fenêtre principale
 * - TerminalPanel : Panel du terminal
 * - TerminalOutput : Gestion de l'affichage avec couleurs
 * 
 * Fonctionnalités :
 * - Affichage du terminal avec couleurs
 * - Historique des commandes (flèches haut/bas)
 * - Auto-complétion (Tab)
 * - Défilement automatique
 * 
 * 6. TESTS
 * -------
 * 
 * Tests unitaires (JUnit 5) :
 * - FileSystemTest : Tests du système de fichiers
 * - DirectoryTest : Tests de la classe Directory
 * - CommandParserTest : Tests du parseur de commandes
 * - LsCommandTest : Tests de la commande ls
 * - CdCommandTest : Tests de la commande cd
 * - CommandFactoryTest : Tests de la factory de commandes
 * - TerminalStateTest : Tests de l'état du terminal
 * - ScriptEngineTest : Tests du moteur de scripts
 * 
 * Tests d'intégration :
 * - IntegrationTest : Tests de l'interaction entre composants
 * 
 * 7. PATTERNS UTILISÉS
 * --------------------
 * 
 * - Command : Pour les commandes du terminal
 * - Factory : Pour la création des commandes (CommandFactory)
 * - Composite : Pour la structure du système de fichiers
 * - State : Pour l'état du terminal (TerminalState)
 * - Strategy : Pour les différents styles d'affichage
 * 
 * 8. EXTENSIBILITÉ
 * ---------------
 * 
 * Pour ajouter une nouvelle commande :
 * 1. Créer une classe implémentant Command
 * 2. L'enregistrer dans CommandFactory
 * 
 * Pour ajouter une nouvelle fonctionnalité au script :
 * 1. Modifier ScriptEngine pour supporter la nouvelle syntaxe
 * 2. Ajouter les méthodes nécessaires
 * 
 * Pour modifier l'interface graphique :
 * 1. Modifier les classes dans le package gui
 * 2. Respecter le pattern MVC
 * 
 * 9. PERFORMANCES
 * --------------
 * 
 * Le système est optimisé pour :
 * - La navigation rapide dans le système de fichiers
 * - L'exécution efficace des commandes
 * - La gestion mémoire des fichiers
 * - Le rendu fluide de l'interface
 * 
 * 10. LIMITATIONS ET AMÉLIORATIONS POSSIBLES
 * ----------------------------------------
 * 
 * Limitations actuelles :
 * - Pas de support des permissions avancées
 * - Pas de support des liens symboliques
 * - Pas de support des tubes (pipes)
 * - Pas de support des redirections avancées
 * - Le script engine est basique
 * 
 * Améliorations possibles :
 * - Ajouter plus de commandes Unix
 * - Améliorer le parseur de scripts
 * - Ajouter le support des couleurs ANSI
 * - Ajouter le support des raccourcis clavier
 * - Ajouter la persistance du système de fichiers
 * - Ajouter le support des plugins
 * 
 * 11. EXEMPLES D'UTILISATION
 * -------------------------
 * 
 * Exemple 1 : Navigation de base
 * ```
 * user@terminal:~$ ls
 * user@terminal:~$ cd documents
 * user@terminal:/home/documents$ pwd
 * /home/documents
 * ```
 * 
 * Exemple 2 : Manipulation de fichiers
 * ```
 * user@terminal:~$ touch test.txt
 * user@terminal:~$ echo "Hello" > test.txt
 * user@terminal:~$ cat test.txt
 * Hello
 * user@terminal:~$ cp test.txt test2.txt
 * user@terminal:~$ mv test.txt renamed.txt
 * user@terminal:~$ rm renamed.txt
 * ```
 * 
 * Exemple 3 : Script simple
 * ```
 * #!/bin/sh
 * set name=World
 * echo Hello $name
 * if [ -f "test.txt" ]
 *   echo File exists
 * fi
 * ```
 * 
 * 12. COMPATIBILITÉ
 * -----------------
 * 
 * - Java 8 ou supérieur
 * - Swing (inclus dans Java SE)
 * - JUnit 5 pour les tests
 * 
 * 13. DÉPENDANCES
 * ---------------
 * 
 * Aucune dépendance externe n'est requise.
 * Le projet utilise uniquement les bibliothèques standard de Java.
 * 
 * 14. COMPILATION ET EXÉCUTION
 * -----------------------------
 * 
 * Compilation :
 * ```bash
 * javac -d bin src/com/terminal/emulator/**/*.java
 * ```
 * 
 * Exécution :
 * ```bash
 * java -cp bin com.terminal.emulator.Main
 * ```
 * 
 * Tests :
 * ```bash
 * javac -d bin -cp .:junit-jupiter-api-5.x.jar:junit-platform-console-standalone-1.x.jar src/**/*.java
 * java -jar junit-platform-console-standalone-1.x.jar --class-path bin --scan-class-path
 * ```
 *
