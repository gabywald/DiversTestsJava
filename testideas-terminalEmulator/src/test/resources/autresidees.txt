package fr.terminal;

import javax.swing.*;
import javax.swing.text.*;
import java.awt.*;
import java.awt.event.*;

/**
 * Classe principale pour l'interface graphique de l'émulateur de terminal.
 * Utilise Swing pour afficher une console interactive.
 */
public class TerminalEmulator {
    private JFrame frame;
    private JTextPane outputArea;
    private JTextField inputField;
    private FileSystem fileSystem;
    private CommandParser commandParser;
    private String currentDirectory;

    /**
     * Constructeur : initialise l'interface graphique et le système de fichiers.
     */
    public TerminalEmulator() {
        fileSystem = new FileSystem();
        commandParser = new CommandParser(fileSystem);
        currentDirectory = fileSystem.getRoot();
        initializeUI();
    }

    /**
     * Initialise l'interface utilisateur (UI).
     */
    private void initializeUI() {
        frame = new JFrame("Terminal Emulator");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setSize(800, 600);
        frame.setLayout(new BorderLayout());

        // Zone de sortie (non éditable)
        outputArea = new JTextPane();
        outputArea.setEditable(false);
        outputArea.setBackground(Color.BLACK);
        outputArea.setForeground(Color.GREEN);
        outputArea.setFont(new Font("Monospaced", Font.PLAIN, 14));
        JScrollPane scrollPane = new JScrollPane(outputArea);
        frame.add(scrollPane, BorderLayout.CENTER);

        // Champ de saisie
        inputField = new JTextField();
        inputField.setBackground(Color.BLACK);
        inputField.setForeground(Color.GREEN);
        inputField.setFont(new Font("Monospaced", Font.PLAIN, 14));
        inputField.addActionListener(e -> executeCommand());
        frame.add(inputField, BorderLayout.SOUTH);

        // Afficher le prompt initial
        printPrompt();

        frame.setVisible(true);
    }

    /**
     * Affiche le prompt dans la zone de sortie.
     */
    private void printPrompt() {
        try {
            StyledDocument doc = outputArea.getStyledDocument();
            doc.insertString(doc.getLength(), currentDirectory + " > ", getPromptStyle());
        } catch (BadLocationException ex) {
            ex.printStackTrace();
        }
    }

    /**
     * Retourne le style pour le prompt.
     */
    private Style getPromptStyle() {
        StyleContext sc = StyleContext.getDefaultStyleContext();
        Style style = sc.addStyle("PromptStyle", null);
        StyleConstants.setForeground(style, Color.GREEN);
        StyleConstants.setBold(style, true);
        return style;
    }

    /**
     * Exécute la commande saisie par l'utilisateur.
     */
    private void executeCommand() {
        String command = inputField.getText().trim();
        inputField.setText("");
        
        if (command.isEmpty()) {
            printPrompt();
            return;
        }

        // Affiche la commande dans la zone de sortie
        appendToOutput(command + "\n");

        // Exécute la commande
        String output = commandParser.execute(command, currentDirectory);
        appendToOutput(output + "\n");

        // Met à jour le répertoire courant si nécessaire
        if (command.startsWith("cd ")) {
            currentDirectory = fileSystem.getCurrentDirectory();
        }

        // Affiche le nouveau prompt
        printPrompt();
    }

    /**
     * Ajoute du texte à la zone de sortie.
     */
    private void appendToOutput(String text) {
        try {
            StyledDocument doc = outputArea.getStyledDocument();
            doc.insertString(doc.getLength(), text, null);
            outputArea.setCaretPosition(doc.getLength());
        } catch (BadLocationException ex) {
            ex.printStackTrace();
        }
    }
}

package fr.terminal;

import java.util.*;

/**
 * Classe représentant un système de fichiers simplifié.
 * Gère les fichiers et répertoires sous forme de structure arborescente.
 */
public class FileSystem {
    private static final String ROOT = "/";
    private String currentDirectory;
    private Map<String, Set<String>> directories;
    private Map<String, String> files;

    /**
     * Constructeur : initialise le système de fichiers avec un répertoire racine.
     */
    public FileSystem() {
        currentDirectory = ROOT;
        directories = new HashMap<>();
        files = new HashMap<>();
        directories.put(ROOT, new HashSet<>());
    }

    /**
     * Retourne le répertoire racine.
     */
    public String getRoot() {
        return ROOT;
    }

    /**
     * Retourne le répertoire courant.
     */
    public String getCurrentDirectory() {
        return currentDirectory;
    }

    /**
     * Change le répertoire courant.
     */
    public void setCurrentDirectory(String path) {
        if (directories.containsKey(path)) {
            currentDirectory = path;
        }
    }

    /**
     * Crée un nouveau répertoire.
     */
    public boolean mkdir(String dirName) {
        String fullPath = getFullPath(dirName);
        if (directories.containsKey(fullPath)) {
            return false; // Répertoire déjà existant
        }
        directories.put(fullPath, new HashSet<>());
        directories.get(currentDirectory).add(dirName);
        return true;
    }

    /**
     * Crée un nouveau fichier.
     */
    public boolean touch(String fileName) {
        String fullPath = getFullPath(fileName);
        if (files.containsKey(fullPath)) {
            return false; // Fichier déjà existant
        }
        files.put(fullPath, "");
        return true;
    }

    /**
     * Supprime un fichier ou un répertoire.
     */
    public boolean rm(String name) {
        String fullPath = getFullPath(name);
        if (files.containsKey(fullPath)) {
            files.remove(fullPath);
            return true;
        } else if (directories.containsKey(fullPath)) {
            directories.remove(fullPath);
            directories.get(currentDirectory).remove(name);
            return true;
        }
        return false;
    }

    /**
     * Liste le contenu du répertoire courant.
     */
    public List<String> ls() {
        List<String> contents = new ArrayList<>();
        Set<String> dirContents = directories.getOrDefault(currentDirectory, new HashSet<>());
        for (String item : dirContents) {
            String fullPath = getFullPath(item);
            if (directories.containsKey(fullPath)) {
                contents.add(item + "/");
            } else if (files.containsKey(fullPath)) {
                contents.add(item);
            }
        }
        Collections.sort(contents);
        return contents;
    }

    /**
     * Lit le contenu d'un fichier.
     */
    public String cat(String fileName) {
        String fullPath = getFullPath(fileName);
        return files.getOrDefault(fullPath, "Fichier introuvable: " + fileName);
    }

    /**
     * Écrit dans un fichier.
     */
    public boolean echo(String fileName, String content) {
        String fullPath = getFullPath(fileName);
        files.put(fullPath, content);
        return true;
    }

    /**
     * Retourne le chemin absolu à partir d'un nom de fichier/répertoire.
     */
    private String getFullPath(String name) {
        if (currentDirectory.equals(ROOT)) {
            return ROOT + name;
        }
        return currentDirectory + "/" + name;
    }

    /**
     * Vérifie si un fichier ou répertoire existe.
     */
    public boolean exists(String name) {
        String fullPath = getFullPath(name);
        return files.containsKey(fullPath) || directories.containsKey(fullPath);
    }

    /**
     * Retourne le chemin absolu du répertoire parent.
     */
    public String getParentDirectory(String path) {
        if (path.equals(ROOT)) {
            return ROOT;
        }
        String[] parts = path.split("/");
        if (parts.length == 2) {
            return ROOT;
        }
        StringBuilder parentPath = new StringBuilder();
        for (int i = 1; i < parts.length - 1; i++) {
            parentPath.append("/").append(parts[i]);
        }
        return parentPath.length() == 0 ? ROOT : parentPath.toString();
    }
}

package fr.terminal;

import fr.terminal.commands.*;
import java.util.*;

/**
 * Classe responsable de l'analyse et de l'exécution des commandes.
 */
public class CommandParser {
    private FileSystem fileSystem;
    private Map<String, Command> commands;

    /**
     * Constructeur : initialise les commandes disponibles.
     */
    public CommandParser(FileSystem fileSystem) {
        this.fileSystem = fileSystem;
        commands = new HashMap<>();
        initializeCommands();
    }

    /**
     * Initialise les commandes disponibles.
     */
    private void initializeCommands() {
        commands.put("ls", new LsCommand(fileSystem));
        commands.put("cd", new CdCommand(fileSystem));
        commands.put("cat", new CatCommand(fileSystem));
        commands.put("echo", new EchoCommand(fileSystem));
        commands.put("mkdir", new MkdirCommand(fileSystem));
        commands.put("touch", new TouchCommand(fileSystem));
        commands.put("rm", new RmCommand(fileSystem));
        commands.put("pwd", new PwdCommand(fileSystem));
        commands.put("help", new HelpCommand());
        commands.put("clear", new ClearCommand());
    }

    /**
     * Exécute une commande.
     */
    public String execute(String input, String currentDirectory) {
        String[] parts = input.split("\\s+");
        if (parts.length == 0) {
            return "";
        }

        String commandName = parts[0].toLowerCase();
        String[] args = Arrays.copyOfRange(parts, 1, parts.length);

        Command command = commands.get(commandName);
        if (command != null) {
            return command.execute(args);
        }
        return "Commande introuvable: " + commandName;
    }
}

package fr.terminal.commands;

import fr.terminal.FileSystem;

/**
 * Commande 'ls' : liste le contenu du répertoire courant.
 */
public class LsCommand implements Command {
    private FileSystem fileSystem;

    public LsCommand(FileSystem fileSystem) {
        this.fileSystem = fileSystem;
    }

    @Override
    public String execute(String[] args) {
        List<String> contents = fileSystem.ls();
        if (contents.isEmpty()) {
            return "";
        }
        return String.join(" ", contents);
    }

package fr.terminal.commands;

import fr.terminal.FileSystem;

/**
 * Commande 'cd' : change le répertoire courant.
 */
public class CdCommand implements Command {
    private FileSystem fileSystem;

    public CdCommand(FileSystem fileSystem) {
        this.fileSystem = fileSystem;
    }

    @Override
    public String execute(String[] args) {
        if (args.length == 0) {
            return "Usage: cd <directory>";
        }

        String targetDir = args[0];
        if (targetDir.equals("..")) {
            String parentDir = fileSystem.getParentDirectory(fileSystem.getCurrentDirectory());
            fileSystem.setCurrentDirectory(parentDir);
            return "";
        } else if (targetDir.equals("/")) {
            fileSystem.setCurrentDirectory(fileSystem.getRoot());
            return "";
        } else {
            String currentDir = fileSystem.getCurrentDirectory();
            String fullPath = currentDir.equals("/") ? "/" + targetDir : currentDir + "/" + targetDir;
            if (fileSystem.exists(targetDir)) {
                fileSystem.setCurrentDirectory(fullPath);
                return "";
            }
            return "Répertoire introuvable: " + targetDir;
        }
    }
}

package fr.terminal.commands;

import fr.terminal.FileSystem;

/**
 * Commande 'echo' : écrit du texte dans un fichier.
 */
public class EchoCommand implements Command {
    private FileSystem fileSystem;

    public EchoCommand(FileSystem fileSystem) {
        this.fileSystem = fileSystem;
    }

    @Override
    public String execute(String[] args) {
        if (args.length < 2) {
            return "Usage: echo <text> > <filename>";
        }

        // Recherche de l'opérateur '>'
        int redirectIndex = -1;
        for (int i = 0; i < args.length; i++) {
            if (args[i].equals(">")) {
                redirectIndex = i;
                break;
            }
        }

        if (redirectIndex == -1 || redirectIndex == args.length - 1) {
            return "Usage: echo <text> > <filename>";
        }

        StringBuilder text = new StringBuilder();
        for (int i = 0; i < redirectIndex; i++) {
            text.append(args[i]).append(" ");
        }
        String fileName = args[redirectIndex + 1];
        fileSystem.echo(fileName, text.toString().trim());
        return "";
    }
}

package fr.terminal;

import org.junit.jupiter.api.*;
import static org.junit.jupiter.api.Assertions.*;

class TerminalEmulatorIntegrationTest {
    private FileSystem fileSystem;
    private CommandParser commandParser;

    @BeforeEach
    void setUp() {
        fileSystem = new FileSystem();
        commandParser = new CommandParser(fileSystem);
    }

    @Test
    void testFullWorkflow() {
        // 1. Créer un répertoire
        commandParser.execute("mkdir test", "/");
        assertTrue(fileSystem.exists("test"));

        // 2. Changer de répertoire
        commandParser.execute("cd test", "/");
        assertEquals("/test", fileSystem.getCurrentDirectory());

        // 3. Créer un fichier
        commandParser.execute("touch file.txt", "/test");
        assertTrue(fileSystem.exists("file.txt"));

        // 4. Écrire dans le fichier
        commandParser.execute("echo Hello > file.txt", "/test");
        assertEquals("Hello", fileSystem.cat("file.txt"));

        // 5. Lister le contenu
        String lsOutput = commandParser.execute("ls", "/test");
        assertTrue(lsOutput.contains("file.txt"));

        // 6. Supprimer le fichier
        commandParser.execute("rm file.txt", "/test");
        assertFalse(fileSystem.exists("file.txt"));

        // 7. Retourner à la racine
        commandParser.execute("cd ..", "/test");
        assertEquals("/", fileSystem.getCurrentDirectory());

        // 8. Supprimer le répertoire
        commandParser.execute("rm test", "/");
        assertFalse(fileSystem.exists("test"));
    }

    @Test
    void testErrorHandling() {
        // Commande inconnue
        String output = commandParser.execute("unknown", "/");
        assertTrue(output.contains("Commande introuvable"));

        // cd vers un répertoire inexistant
        output = commandParser.execute("cd nonexistent", "/");
        assertTrue(output.contains("Répertoire introuvable"));

        // cat sur un fichier inexistant
        output = commandParser.execute("cat nonexistent.txt", "/");
        assertTrue(output.contains("Fichier introuvable"));
    }
}
